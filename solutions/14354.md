# Id 14354 — Sort Characters by Frequency

```c
#include <stdio.h>
#include <string.h>

int main() {
    char s[1001];
    scanf("%s", s);
    int cnt[256] = {0};
    char order[256];
    int n = 0, i, j;
    for (i = 0; s[i]; i++) {
        unsigned char c = (unsigned char)s[i];
        if (cnt[c] == 0) order[n++] = (char)c;
        cnt[c]++;
    }
    /* stable insertion sort by count descending (ties keep first-occurrence order) */
    for (i = 1; i < n; i++) {
        char key = order[i];
        j = i - 1;
        while (j >= 0 && cnt[(unsigned char)order[j]] < cnt[(unsigned char)key]) {
            order[j + 1] = order[j];
            j--;
        }
        order[j + 1] = key;
    }
    for (i = 0; i < n; i++) printf("%c", order[i]);
    printf("\n");
    return 0;
}
```
