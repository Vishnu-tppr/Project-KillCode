# Id 14182 — String - Alphabets, Digits or Both

```c
#include <stdio.h>
#include <ctype.h>

int main() {
    char s[101];
    scanf("%s", s);
    int al = 0, nu = 0, i;
    for (i = 0; s[i]; i++) {
        if (isalpha((unsigned char) s[i])) al = 1;
        else if (isdigit((unsigned char) s[i])) nu = 1;
    }
    if (al && nu) printf("alphanumeric\n");
    else if (al) printf("alpha\n");
    else printf("numeric\n");
    return 0;
}
```
