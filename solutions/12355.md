# Id 12355 — Most Frequent Alphabet - Substrings

```c
#include <stdio.h>
#include <string.h>

int main() {
    char s[1001];
    int q;
    scanf("%s", s);
    scanf("%d", &q);
    int n = (int) strlen(s);
    int pre[1001][26];
    int i, k;
    for (k = 0; k < 26; k++)
        pre[0][k] = 0;
    for (i = 0; i < n; i++) {
        for (k = 0; k < 26; k++)
            pre[i + 1][k] = pre[i][k];
        pre[i + 1][s[i] - 'a']++;
    }
    while (q--) {
        int l, r;
        scanf("%d %d", &l, &r);
        int best = 0, bestc = -1;
        for (k = 0; k < 26; k++) {
            int c = pre[r + 1][k] - pre[l][k];
            if (c > bestc) {
                bestc = c;
                best = k;
            }
        }
        putchar('a' + best);
        putchar('\n');
    }
    return 0;
}
```
