# Id 6698 — Even or Odd Integers

```c
#include <stdio.h>

int main() {
    int a, b, c, x;
    scanf("%d%d%d%d", &a, &b, &c, &x);
    int e = 0;
    if (a % 2 == 0) e++;
    if (b % 2 == 0) e++;
    if (c % 2 == 0) e++;
    int f = 0;
    if (e >= 2) {
        for (int i = 2; i <= x; i += 2) {
            if (f) printf(" ");
            f = 1;
            printf("%d", i);
        }
    } else {
        for (int i = x; i >= 1; i -= 2) {
            if (f) printf(" ");
            f = 1;
            printf("%d", i);
        }
    }
    printf("\n");
    return 0;
}
```

Verified: `2 4 5 10 → 2 4 6 8 10`, `11 3 6 15 → 15 13 11 9 7 5 3 1`
