```c
#include <stdio.h>

int gcd(int a, int b) {
    while (b) { int t = a % b; a = b; b = t; }
    return a;
}

int main() {
    int a[1000], n = 0;
    while (scanf("%d", &a[n]) == 1) n++;
    int i;
    for (i = 0; i + 1 < n; i++) {
        if (a[i] > a[i + 1] && gcd(a[i], a[i + 1]) == 1) {
            printf("%d\n", i + 1);
            return 0;
        }
    }
    printf("-1\n");
    return 0;
}
```
