# Id 7109 — Unique Digit Sum Count

```c
#include <stdio.h>

int main() {
    int n;
    scanf("%d", &n);
    int seen[200] = {0};
    int count = 0;
    for (int i = 0; i < n; i++) {
        long long x;
        scanf("%lld", &x);
        int ds = 0;
        while (x) { ds += x % 10; x /= 10; }
        if (!seen[ds]) { seen[ds] = 1; count++; }
    }
    printf("%d\n", count);
    return 0;
}

```
