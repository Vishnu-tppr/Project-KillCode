# Id 14468 — Power of 2

```c
#include <stdio.h>

int main() {
    long long N;
    scanf("%lld", &N);
    long long p = 1;
    while (p < N) p *= 2;
    if (p == N) {
        long long s = 0, t = N;
        while (t) { s += t % 10; t /= 10; }
        printf("Yes\n%lld\n", s);
    } else {
        printf("No\n%lld\n", p);
    }
    return 0;
}
```
