# Id 14377 — Nth Special Number

```c
#include <stdio.h>

long long count_le(long long v) {
    long long p = v / 10, d = v % 10;
    long long c = p * 6;
    if (d <= 5) c += d + 1;
    else c += 6;
    return c;
}

int main() {
    long long N;
    scanf("%lld", &N);
    long long lo = 0, hi = 10 * N + 5; /* upper bound for N-th special number */
    while (lo < hi) {
        long long mid = (lo + hi) / 2;
        if (count_le(mid) >= N) hi = mid;
        else lo = mid + 1;
    }
    printf("%lld\n", lo);
    return 0;
}
```
