# Id 14388 — Number to Open Next Level

```c
#include <stdio.h>

int main() {
    int N;
    scanf("%d", &N);
    int arr[100];
    int i;
    for (i = 0; i < N; i++)
        scanf("%d", &arr[i]);
    int X;
    scanf("%d", &X);
    long long cand = X + 1;
    while (1) {
        int ok = 0;
        for (i = 0; i < N; i++) {
            if (cand % arr[i] == 0) {
                ok = 1;
                break;
            }
        }
        if (ok) {
            printf("%lld\n", cand);
            return 0;
        }
        cand++;
    }
    return 0;
}
```
