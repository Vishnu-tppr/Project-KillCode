# Id 5413 — C - Function - Minimum of N Integers

```c
#include <stdio.h>
void printMin(int N, int arr[]) {
    int mn = arr[0];
    for (int i = 1; i < N; i++)
        if (arr[i] < mn) mn = arr[i];
    printf("%d", mn);
}
int main() {
    int n, a[1001];
    scanf("%d", &n);
    for (int i = 0; i < n; i++) scanf("%d", &a[i]);
    printMin(n, a);
    printf("\n");
    return 0;
}
```
