# Id 14500 — Food Item - Maximum Product

```c
#include <stdio.h>

int main() {
    int N;
    scanf("%d", &N);
    long long arr[100];
    int i;
    for (i = 0; i < N; i++)
        scanf("%lld", &arr[i]);
    long long best = -1;
    int bestIdx = 0;
    for (i = 0; i < N; i++) {
        long long left = arr[(i - 1 + N) % N];
        long long right = arr[(i + 1) % N];
        long long prod = left * right;
        if (prod > best) {
            best = prod;
            bestIdx = i;
        }
    }
    printf("%lld\n", arr[bestIdx]);
    return 0;
}
```
