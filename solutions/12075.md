```c
int countHardPrograms(int N, int *arr, int K)
{
    int page = 1;
    int hard = 0;
    int c, p;
    for (c = 0; c < N; c++)
    {
        for (p = 1; p <= arr[c]; p++)
        {
            if (page + (p - 1) / K == p)
                hard++;
        }
        page += (arr[c] + K - 1) / K;
    }
    return hard;
}
```
