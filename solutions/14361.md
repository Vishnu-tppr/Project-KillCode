int messageEncryptionKey(int numMessage)
{
    int sum = 0;
    int n = numMessage;
    if (n < 0) n = -n;
    while (n)
    {
        int d = n % 10;
        if (d != 0 && d != 1 && d != 2 && d != 3 && d != 5 && d != 7)
