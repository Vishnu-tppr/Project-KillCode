# Id 12343 — Shortest & Second Shortest Distance

```c
#include <stdio.h>

int main() {
    int N;
    scanf("%d", &N);
    long long mn = 100000000000LL, mn2 = 100000000000LL;
    int i;
    for (i = 0; i < N; i++) {
        long long v;
        scanf("%lld", &v);
        if (v < mn) { mn2 = mn; mn = v; }
        else if (v > mn && v < mn2) mn2 = v;
    }
    if (mn2 == 100000000000LL) printf("Equal\n");
    else printf("%lld %lld\n", mn, mn2);
    return 0;
}
```
