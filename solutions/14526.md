# Id 14526 — Packing Balls - Maximum Boxes

```c
#include <stdio.h>

int main() {
    long long R, W, b1, best = 0;
    scanf("%lld %lld", &R, &W);
    for (b1 = 0; 2 * b1 <= R; b1++) {
        long long b2 = R - 2 * b1;
        long long b2w = (W - b1) / 2;
        if (b2w < b2) b2 = b2w;
        if (b2 >= 0 && b1 + b2 > best) best = b1 + b2;
    }
    printf("%lld\n", best);
    return 0;
}
```
