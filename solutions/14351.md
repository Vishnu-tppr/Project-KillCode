int findMaximum(int *charges, char *sign, int N)
{
    long long arr[100000];
    int i;
    for (i = 0; i < N; i++)
        arr[i] = (sign[i] == 'P') ? charges[i] : -(long long)charges[i];
    long long best = arr[0];
    long long cur = arr[0];
    for (i = 1; i < N; i++) {
        if (cur + arr[i] > arr[i])
            cur = cur + arr[i];
        else
            cur = arr[i];
        if (cur > best)
            best = cur;
    }
    return (int)(best * 100);
}
