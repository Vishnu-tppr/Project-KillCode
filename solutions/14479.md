static char resultBuf[2000];

char *findAlphabetsOrDigitSum(char *str, int option)
{
    int i;
    int m = 0;
    if (option == 0) {
        long long sum = 0;
        int hasDigit = 0;
        for (i = 0; str[i] != '\0'; i++) {
            if (str[i] >= '0' && str[i] <= '9') {
                sum += str[i] - '0';
                hasDigit = 1;
            }
        }
        if (!hasDigit) {
            resultBuf[0] = 'Z';
            resultBuf[1] = 'E';
            resultBuf[2] = 'R';
            resultBuf[3] = 'O';
            resultBuf[4] = '\0';
            return resultBuf;
        }
        if (sum == 0) {
            resultBuf[0] = '0';
            resultBuf[1] = '\0';
            return resultBuf;
        }
        char tmp[32];
        int len = 0;
        while (sum > 0) {
            tmp[len++] = (char)('0' + sum % 10);
            sum /= 10;
        }
        m = 0;
        while (len > 0)
            resultBuf[m++] = tmp[--len];
        resultBuf[m] = '\0';
        return resultBuf;
    } else {
        int hasAlpha = 0;
        for (i = 0; str[i] != '\0'; i++) {
            if ((str[i] >= 'A' && str[i] <= 'Z') || (str[i] >= 'a' && str[i] <= 'z')) {
                resultBuf[m++] = str[i];
                hasAlpha = 1;
            }
        }
        if (!hasAlpha) {
            resultBuf[0] = 'Z';
            resultBuf[1] = 'E';
            resultBuf[2] = 'R';
            resultBuf[3] = 'O';
            resultBuf[4] = '\0';
            return resultBuf;
        }
        resultBuf[m] = '\0';
        return resultBuf;
    }
}
