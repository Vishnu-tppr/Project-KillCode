# Id 14235 — Picnic - Maximum Groups

```c
#include <stdio.h>

int main() {
    int N, i, j;
    scanf("%d", &N);
    int x[1000];
    for (i = 0; i < N; i++) scanf("%d", &x[i]);
    int groups = 0;
    i = 0;
    while (i < N) {
        int cnt[51] = {0};
        int formed = 0;
        j = i;
        while (j < N) {
            cnt[x[j]]++;
            if (cnt[x[j]] >= 2) {
                formed = 1;
                break;
            }
            j++;
        }
        if (formed) {
            groups++;
            i = j + 1;
        } else {
            i++;
        }
    }
    printf("%d\n", groups);
    return 0;
}
```
