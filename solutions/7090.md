# Id 7090 — Cumulative Sum of Prime Integers

```c
#include <stdio.h>

int isPrime(int n) {
    if (n < 2) return 0;
    for (int i = 2; i * i <= n; i++) if (n % i == 0) return 0;
    return 1;
}

int main() {
    int n;
    scanf("%d", &n);
    long long sum = 0;
    int cnt = 0, num = 2;
    while (cnt < n) {
        if (isPrime(num)) {
            sum += num;
            if (cnt) printf(" ");
            printf("%lld", sum);
            cnt++;
        }
        num++;
    }
    printf("\n");
    return 0;
}

```
