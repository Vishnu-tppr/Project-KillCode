#include <string.h>

int main() {

    scanf("%d", &X);

    if (X == 0) {

    }
    int L = 0, l = 0;
    for (int r = 0; r < n; r++) {
        while (pref[r + 1] - pref[l] > X) l++;
        if (r - l + 1 > L) L = r - l + 1;
    }
    int cnt = 0;
    for (int i = 0; i + L <= n; i++)
        if (pref[i + L] - pref[i] <= X) cnt++;
    printf("%d\n", cnt);
    return 0;
}
