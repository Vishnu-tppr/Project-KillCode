int findLargestDistance(char *str, char ch)
{
    int pos[2000];
    int n = 0;
    int i;
    for (i = 0; str[i] != '\0'; i++)
        if (str[i] == ch)
            pos[n++] = i;
    if (n < 2)
        return -1;
    int seen[128] = {0};
    int cnt = 0;
    for (i = pos[0] + 1; i < pos[n - 1]; i++) {
        if (str[i] == ' ' || str[i] == ch)
            continue;
        if (!seen[(unsigned char)str[i]]) {
            seen[(unsigned char)str[i]] = 1;
            cnt++;
        }
    }
    return cnt;
}
