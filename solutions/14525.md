#include <stdio.h>

int main() {
    long long N;
    scanf("%lld", &N);
    int digits[12];
    int n = 0;
    while (N > 0) {
        digits[n++] = (int)(N % 10);
        N /= 10;
    }
    int i, j;
    for (i = 0, j = n - 1; i < j; i++, j--) {
        int t = digits[i];
        digits[i] = digits[j];
        digits[j] = t;
    }
    long long count = 0;
    for (i = 0; i < n; i++) {
        if (digits[i] % 2 == 0)
            count += i + 1;
    }
    printf("%lld\n", count);
    return 0;
}
