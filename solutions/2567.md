#include <stdio.h>
int main() {
    long long n;
    scanf("%lld", &n);
    int prime = 1;
    if (n < 2) prime = 0;
    for (long long i = 2; i * i <= n && prime; i++)
        if (n % i == 0) prime = 0;
    printf("%s\n", prime ? "YES" : "NO");
    return 0;
}
