int main() {
    int r, c; scanf("%d %d", &r, &c);
    char m1[100][100], m2[100][100];
    for (int i = 0; i < r; i++)
        for (int j = 0; j < c; j++) scanf(" %c", &m1[i][j]);
    for (int i = 0; i < r; i++)
        for (int j = 0; j < c; j++) scanf(" %c", &m2[i][j]);
    for (int i = 0; i < r; i++) {
        for (int j = 0; j < c; j++) {
            if (j) printf(" ");
            printf("%c", m1[i][j] > m2[i][j] ? m1[i][j] : m2[i][j]);
        }
        printf("\n");
    }
    return 0;
}
