# Id 14462 — Alt+Tab Window

```c
#include <stdlib.h>

int *altTabWindow(int N, int *arr, int T)
{
    int k = ((T - 1) % N + N) % N;
    int *res = (int *)malloc(N * sizeof(int));
    int m = 0;
    int i;
    res[m++] = arr[k];
    for (i = 0; i < N; i++)
        if (i != k)
            res[m++] = arr[i];
    return res;
}
```
