# Id 7123 — Lexicographically in Descending Order

```c
#include <stdio.h>
#include <string.h>
#include <stdlib.h>

char arr[50][1001];

int cmp(const void *a, const void *b) {
    return strcmp((const char*)b, (const char*)a);
}

int main() {
    int n;
    scanf("%d", &n);
    for (int i = 0; i < n; i++) scanf("%s", arr[i]);
    qsort(arr, n, 1001, cmp);
    for (int i = 0; i < n; i++) printf("%s\n", arr[i]);
    return 0;
}

```
