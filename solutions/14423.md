int getMaxRowColSum(int R, int C, int *arr) {
    int i, j;
    long long maxRow = 0, maxCol = 0;
    for (i = 0; i < R; i++) {
        long long s = 0;
        for (j = 0; j < C; j++) s += arr[i * C + j];
        if (s > maxRow) maxRow = s;
    }
    for (j = 0; j < C; j++) {
        long long s = 0;
        for (i = 0; i < R; i++) s += arr[i * C + j];
        if (s > maxCol) maxCol = s;
    }
    return (int)(maxRow + maxCol);
}
