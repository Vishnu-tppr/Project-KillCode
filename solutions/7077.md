# Id 7077 — Middle Character(s)

#include <stdio.h>
#include <string.h>
int main(){
    int N;
    scanf("%d",&N);
    char s[101];
    while(getchar()!='\n');
    for(int i=0;i<N;i++){
        fgets(s,101,stdin);
        s[strcspn(s,"\n")]=0;
        int len=strlen(s);
        if(len%2==1) printf("%c\n",s[len/2]);
        else printf("%c%c\n",s[len/2-1],s[len/2]);
    }
    return 0;
}
