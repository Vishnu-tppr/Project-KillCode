# Id 14486 — Split Chocolates - Equal Weight

```c
#include <stdio.h>

int main() {
    int N;
    scanf("%d", &N);
    int a[16];
    int sum = 0;
    int i;
    for (i = 0; i < N; i++) { scanf("%d", &a[i]); sum += a[i]; }
    if (sum % 2) { printf("NO\n"); return 0; }
    int target = sum / 2;
    static int dp[1505];
    dp[0] = 1;
    for (i = 0; i < N; i++)
        for (int s = target; s >= a[i]; s--)
            if (dp[s - a[i]]) dp[s] = 1;
    printf("%s\n", dp[target] ? "YES" : "NO");
    return 0;
}
```
