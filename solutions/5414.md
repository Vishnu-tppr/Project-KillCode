# Id 5414 — C - Function - Array Elements Sum

```c
#include <stdio.h>
void printArrayElementsSum(int N, int arr[]) {
    int s = 0;
    for (int i = 0; i < N; i++) s += arr[i];
    printf("%d", s);
}
int main() {
    int n, a[1001];
    scanf("%d", &n);
    for (int i = 0; i < n; i++) scanf("%d", &a[i]);
    printArrayElementsSum(n, a);
    printf("\n");
    return 0;
}
```
