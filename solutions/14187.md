# Id 14187 — Integers within Range

```c
#include <stdio.h>

int main() {
    int N;
    scanf("%d", &N);
    int a[100];
    for (int i = 0; i < N; i++) scanf("%d", &a[i]);
    int R1, R2;
    scanf("%d %d", &R1, &R2);
    int cnt = 0;
    for (int i = 0; i < N; i++)
        if (a[i] >= R1 && a[i] <= R2) cnt++;
    printf("%d\n", cnt);
    return 0;
}
```
