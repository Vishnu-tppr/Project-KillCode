#include <stdio.h>
#include <string.h>

int main() {
    int N, i, X;
    char st[31][3];
    int d[31];
    char s[3], t[3];
    scanf("%d", &N);
    for (i = 0; i < N; i++) scanf("%s", st[i]);
    for (i = 0; i < N; i++) scanf("%d", &d[i]);
    scanf("%s %s", s, t);
    scanf("%d", &X);
    int si = -1, ti = -1;
    for (i = 0; i < N; i++) {
        if (strcmp(st[i], s) == 0) si = i;
        if (strcmp(st[i], t) == 0) ti = i;
    }
    long long tot = 0;
    int cur = si;
    while (cur != ti) {
        int nx = (cur + 1) % N;
        tot += d[nx];
        cur = nx;
    }
    long long cost = (tot * X + 999) / 1000;
    printf("%lld\n", cost);
    return 0;
}
