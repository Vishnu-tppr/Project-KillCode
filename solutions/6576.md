# Id 6576 — Area of a Ground

```c
#include <stdio.h>
int main() {
    long long a, b, h;
    scanf("%lld%lld%lld", &a, &b, &h);
    long long t = (a + b) * h;
    if (t % 2 == 0) printf("%lld.00\n", t / 2);
    else printf("%lld.5\n", t / 2);
    return 0;
}
```
