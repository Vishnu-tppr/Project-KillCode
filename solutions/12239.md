# Id 12239 — Exam Seating Arrangement

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

int main() {
    int M, N;
    scanf("%d %d", &M, &N);
    int cols = M / N;
    char grid[101][101][8];
    int i, j;
    for (i = 0; i < N; i++)
        for (j = 0; j < cols; j++)
            strcpy(grid[i][j], "ABSENT");
    char s[16];
    for (i = 0; i < M; i++) {
        scanf("%s", s);
        int isXII = (s[1] == 'I');
        int num = atoi(s + (isXII ? 3 : 1));
        int col = ((num - 1) / N) * 2 + (isXII ? 2 : 1);
        int row = (num - 1) % N;
        strcpy(grid[row][col - 1], s);
    }
    for (i = 0; i < N; i++) {
        for (j = 0; j < cols; j++) {
            if (j) printf(" ");
            printf("%s", grid[i][j]);
        }
        printf("\n");
    }
    return 0;
}
```
