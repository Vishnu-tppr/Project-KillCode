#include <stdio.h>
#include <stdlib.h>
#include <string.h>

int main() {
    char s[16];
    scanf("%s", s);
    int neg = 0;
    if (s[0] == '-') { neg = 1; memmove(s, s + 1, strlen(s)); }
    int len = strlen(s);
    int found = 0;
    long long best = 0;
    int i, j;
    for (i = 0; i < len; i++) {
        if (s[i] != '5') continue;
        char t[16];
        int k = 0;
        for (j = 0; j < len; j++)
            if (j != i) t[k++] = s[j];
        t[k] = 0;
        long long v = atoll(t);
        if (neg) v = -v;
        if (!found || v > best) { best = v; found = 1; }
    }
    printf("%lld\n", best);
    return 0;
}
