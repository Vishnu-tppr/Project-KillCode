#include <stdio.h>
#include <string.h>

int main() {
    char s[1001];
    scanf("%s", s);
    int f[26] = {0};
    int len = strlen(s);
    for (int i = 0; i < len; i++) f[s[i]-'a']++;
    int target = 0;
    for (int i = 0; i < 26; i++)
        if (f[i]) { target = f[i]; break; }
    for (int i = 0; i < 26; i++)
        if (f[i] && f[i] != target) { printf("NO\n"); return 0; }
    printf("YES\n");
    return 0;
}
