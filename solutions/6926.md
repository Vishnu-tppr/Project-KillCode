# Id 6926 — Even Integers in Descending Order

```c
#include <stdio.h>
int main(){
    int n; int a[50],e[50],cnt=0;
    int i,j;
    scanf("%d",&n);
    for(i=0;i<n;i++){scanf("%d",&a[i]);
        if(a[i]%2==0)e[cnt++]=a[i];
    }
    for(i=0;i<cnt;i++)
        for(j=i+1;j<cnt;j++)
            if(e[j]>e[i]){int t=e[i];e[i]=e[j];e[j]=t;}
    for(i=0;i<cnt;i++){if(i>0)printf(" ");printf("%d",e[i]);}
    printf("\n");
    return 0;
}
```
