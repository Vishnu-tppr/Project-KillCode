#include <stdlib.h>
#include <string.h>

int main() {
    char line1[2048], line2[2048], ch;

    int items[1000], qtys[1000], ni = 0, nq = 0;
    char *p = strtok(line1, " \n");
    while (p) { items[ni++] = atoi(p); p = strtok(NULL, " \n"); }
    p = strtok(line2, " \n");
    while (p) { qtys[nq++] = atoi(p); p = strtok(NULL, " \n"); }
    int price[6] = {0, 180, 120, 50, 70, 90};
    long long total = 0;
    int i;
    for (i = 0; i < ni; i++) total += (long long)price[items[i]] * qtys[i];
    int disc = (total > 1000 ? 10 : 0) + (ch == 'Y' ? 5 : 0);
    double final = total * (100 - disc) / 100.0;
    printf("%.2f\n", final);
    return 0;
}
