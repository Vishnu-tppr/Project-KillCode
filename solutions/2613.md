# Id 2613 — String - Reverse Words [ZOHO]

```c
#include <stdio.h>
#include <string.h>
int main() {
    char s[101];
    char *words[100];
    int n = 0;
    fgets(s, sizeof(s), stdin);
    char *tok = strtok(s, " \n");
    while (tok) {
        words[n++] = tok;
        tok = strtok(NULL, " \n");
    }
    for (int i = n - 1; i >= 0; i--) {
        printf("%s", words[i]);
        if (i) printf(" ");
    }
    printf("\n");
    return 0;
}
```
