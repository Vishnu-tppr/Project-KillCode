# Id 14502 — Find Coupon Code - Winners

```c
#include <stdio.h>

int main() {
    int N;
    scanf("%d", &N);
    int freq[1001] = {0};
    for (int i = 0; i < N; i++) {
        int v;
        scanf("%d", &v);
        freq[v]++;
    }
    int ans = -1;
    for (int v = 1; v <= 1000; v++)
        if (freq[v] == v) ans = v;
    printf("%d\n", ans);
    return 0;
}
```
