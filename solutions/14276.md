# Id 14276 — Minimum Penalty Absolute Diff

```c
int minimumPenalty(int N, int arr[])
{
    int i, j;
    for (i = 0; i < N - 1; i++)
        for (j = i + 1; j < N; j++)
            if (arr[j] < arr[i]) { int t = arr[i]; arr[i] = arr[j]; arr[j] = t; }
    long long sum = 0;
    for (i = 1; i < N; i++)
    {
        long long d = arr[i] - arr[i - 1];
        if (d < 0) d = -d;
        sum += d;
    }
    return (int)sum;
}
```
