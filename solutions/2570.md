# Id 2570 — Uppercase Letters Count

```c
#include <stdio.h>
#include <string.h>
int main() {
    char s[101];
    int count = 0;
    fgets(s, sizeof(s), stdin);
    s[strcspn(s, "\n")] = '\0';
    for (int i = 0; s[i]; i++)
        if (s[i] >= 'A' && s[i] <= 'Z') count++;
    printf("%d\n", count);
    return 0;
}
```
