# Id 12283 — Count Characters & Words

```c
#include <stdio.h>
#include <string.h>

int main() {
    char s[1000];
    if (fgets(s, sizeof(s), stdin) == NULL)
        return 0;
    int len = (int)strlen(s);
    while (len > 0 && (s[len - 1] == '\n' || s[len - 1] == '\r'))
        s[--len] = '\0';
    int chars = len;
    int words = 0;
    int i = 0;
    while (i < chars) {
        while (i < chars && s[i] == ' ')
            i++;
        if (i < chars) {
            words++;
            while (i < chars && s[i] != ' ')
                i++;
        }
    }
    printf("%d\n%d\n", chars, words);
    return 0;
}
```
