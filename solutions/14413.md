# Id 14413 — Two Highest Selling Products

#include <stdio.h>

int main() {
    int n, i;
    scanf("%d", &n);
    long long m1 = -1, m2 = -1;
    for (i = 0; i < n; i++) {
        long long v;
        scanf("%lld", &v);
        if (v > m1) {
            m2 = m1;
            m1 = v;
        } else if (v > m2) {
            m2 = v;
        }
    }
    printf("%lld\n", m1 + m2);
    return 0;
}
