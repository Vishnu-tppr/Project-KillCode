# Id 6835 — Adjacent Characters

```c
#include <stdio.h>
#include <string.h>
int main(){
    char s[102]; char ch;
    scanf("%s %c",s,&ch);
    int len=strlen(s); int found=0;
    for(int i=0;i<len;i++){
        if(s[i]==ch){
            if(i>0){putchar(s[i-1]);found=1;}
            if(i<len-1){putchar(s[i+1]);found=1;}
        }
    }
    if(!found)printf("-1");
    printf("\n");
    return 0;
}
```
