void validateMessage(char *message)
{
    int L = 0;
    while (message[L]) L++;
    int ok = 1;
    for (int i = 0; i < L; i++) {
        char c = message[i];
        if (!((c >= 'A' && c <= 'Z') || (c >= 'a' && c <= 'z'))) { ok = 0; break; }
    }
    if (ok) {
        for (int i = 0; i < L; i++) {
            if (message[i] != message[L - 1 - i]) { ok = 0; break; }
        }
    }
    if (ok) printf("Strong Message\n");
    else printf("Weak Message\n");
}
