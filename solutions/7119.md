# Id 7119 — Camel Case String

```c
#include <stdio.h>
#include <ctype.h>

int main() {
    char s[101];
    scanf("%s", s);
    for (int i = 0; s[i]; i++) {
        if (i > 0 && isupper(s[i])) printf(" ");
        printf("%c", tolower(s[i]));
    }
    printf("\n");
    return 0;
}
```
