    int N;

        scanf("%lld", &arr[i]);
    int cut = N;
    int steps = 0;
    while (cut > 0) {
        int maxIdx = 0;
        for (i = 1; i < cut; i++)
            if (arr[i] > arr[maxIdx])

    }

}
