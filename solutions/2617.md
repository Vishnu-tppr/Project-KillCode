# Id 2617 — String - Count Articles

```c
#include <stdio.h>
#include <string.h>
int main() {
    char s[1001];
    fgets(s, sizeof(s), stdin);
    int count = 0;
    char *tok = strtok(s, " \t\n.,;:!?()\"'-");
    while (tok) {
        if (strcmp(tok, "a") == 0 || strcmp(tok, "an") == 0 || strcmp(tok, "the") == 0)
            count++;
        tok = strtok(NULL, " \t\n.,;:!?()\"'-");
    }
    printf("%d\n", count);
    return 0;
}
```
