# Id 12541 — Digits Sum Starts with D

```c
#include <stdio.h>

int main() {
    int N, D;
    scanf("%d %d", &N, &D);
    int first = 1;
    for (int i = 1; i <= N; i++) {
        int x = i, s = 0;
        while (x > 0) { s += x % 10; x /= 10; }
        int lead = (s >= 10) ? (s / 10) : s;
        if (lead == D) {
            if (first) { printf("%d", i); first = 0; }
            else printf(" %d", i);
        }
    }
    printf("\n");
    return 0;
}
```
