#include <stdio.h>
#include <stdlib.h>

int cmp(const void *a, const void *b) { return (*(int *)a - *(int *)b); }

int main() {
    int N, R, i;
    int a[1000];
    scanf("%d%d", &N, &R);
    for (i = 0; i < N; i++) scanf("%d", &a[i]);
    qsort(a, N, sizeof(int), cmp);
    int count = 0;
    i = 0;
    while (i < N) {
        count++;
        long long limit = (long long)a[i] + 2 * R;
        while (i < N && a[i] <= limit) i++;
    }
    printf("%d\n", count);
    return 0;
}
