# Id 14495 — Count Houses - Electricity

```c
#include <stdio.h>

int main() {
    int N, V, i, cnt = 0;
    scanf("%d%d", &N, &V);
    for (i = 1; i <= N; i++) {
        if (V % i == 0) {
            int j = V / i;
            if (j >= 1 && j <= N) cnt++;
        }
    }
    if (cnt == 0) printf("-1\n");
    else printf("%d\n", cnt);
    return 0;
}
```
