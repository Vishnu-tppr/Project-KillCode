# Id 12333 — Count of Non-repeated Characters

```c
#include <stdio.h>

int main() {
    char s[1001];
    scanf("%s", s);
    int cnt[256] = {0};
    int i;
    for (i = 0; s[i]; i++) cnt[(unsigned char)s[i]]++;
    int c = 0;
    for (i = 0; i < 256; i++)
        if (cnt[i] == 1) c++;
    printf("%d\n", c);
    return 0;
}
```
