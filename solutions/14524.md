#include <string.h>

int main() {
    char s[16];
    scanf("%s", s);
    int len = strlen(s);
    int i;
    for (i = 0; i + 1 < len; i++) {
        int d = s[i] > s[i + 1] ? s[i] - s[i + 1] : s[i + 1] - s[i];
