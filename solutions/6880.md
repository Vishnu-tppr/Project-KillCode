# Id 6880 — Odd or Even Length of S

```c
#include <stdio.h>
#include <string.h>
int main(){
    char s[101];
    fgets(s,101,stdin);
    s[strcspn(s,"\n")]=0;
    int len=strlen(s);
    if(len%2==1){
        for(int i=len-1;i>=0;i--)putchar(s[i]);
    }else{
        printf("%s",s);
    }
    printf("\n");
    return 0;
}
```
