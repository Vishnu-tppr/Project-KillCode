# Id 12284 — Walls Reconstruction - Strictly Increasing Order

```c
#include <stdio.h>

int main() {
    int N, i;
    scanf("%d", &N);
    long long a[100], total = 0;
    for (i = 0; i < N; i++) scanf("%lld", &a[i]);
    for (i = 0; i < N; i++) {
        if (i > 0 && a[i] <= a[i - 1]) a[i] = a[i - 1] + 1;
        total += a[i];
    }
    printf("%lld\n", total);
    return 0;
}
```
