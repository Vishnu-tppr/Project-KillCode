char S[11];
int N, len;
long long best = -1;

void solve(int start, int parts, long long curmax) {
    if (parts == N) return;
    if (parts == N - 1) {
        long long val = 0;
        int j;
        for (j = start; j < len; j++) val = val * 10 + (S[j] - '0');
        if (val > curmax) curmax = val;
        if (curmax > best) best = curmax;
        return;
    }
    int i;
    for (i = start; i <= len - N + parts; i++) {
        long long val = 0;
        int j;
        for (j = start; j <= i; j++) val = val * 10 + (S[j] - '0');
        long long nm = val > curmax ? val : curmax;
        solve(i + 1, parts + 1, nm);
    }
}

int main() {

    scanf("%d", &N);
    len = strlen(S);

        long long mx = 0;
        int j;
        for (j = 0; j < len; j++) {
            long long d = S[j] - '0';
            if (d > mx) mx = d;
        }
        printf("%lld\n", mx);
        return 0;
    }
    solve(0, 0, 0);
    printf("%lld\n", best);
    return 0;
}
