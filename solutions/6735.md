# Id 6735 — N Multiples of X

```c
#include <stdio.h>
int main(){
    long long N,X,Y;
    long long i,count=0;
    scanf("%lld %lld %lld",&N,&X,&Y);
    for(i=1;i<=N;i++){
        long long m=X*i;
        if(m%Y!=0){if(count>0)printf(" ");printf("%lld",m);count++;}
    }
    printf("\n");
    return 0;
}
```
