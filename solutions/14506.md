#include <stdio.h>
#include <string.h>

int isVowel(char c) {
    if (c >= 'A' && c <= 'Z')
        c += 32;
    return c == 'a' || c == 'e' || c == 'i' || c == 'o' || c == 'u';
}

int main() {
    char s[1001];
    scanf("%s", s);
    int n = (int)strlen(s);
    int count = 0;
    int i;
    for (i = 1; i < n; i++) {
        if (!isVowel(s[i]) && isVowel(s[i - 1]))
            count++;
    }
    printf("%d\n", count);
    return 0;
}
