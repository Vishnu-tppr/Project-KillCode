# Id 12339 — Count Unique Words

```c
#include <stdio.h>
#include <string.h>

int main() {
    char s[1001];
    fgets(s, sizeof(s), stdin);
    char *words[500];
    int n = 0, i, j;
    char *tok = strtok(s, " \n\t");
    while (tok) {
        words[n++] = tok;
        tok = strtok(NULL, " \n\t");
    }
    int count = 0;
    for (i = 0; i < n; i++) {
        int times = 0;
        for (j = 0; j < n; j++) {
            if (strcmp(words[i], words[j]) == 0) times++;
        }
        if (times == 1) count++;
    }
    printf("%d\n", count);
    return 0;
}
```
