#include <stdio.h>

int main() {
    int n, i;
    scanf("%d", &n);
    long long v[100000];
    for (i = 0; i < n; i++)
        scanf("%lld", &v[i]);

    int i1 = 0, i2 = -1, i3 = 0, i4 = -1;
    for (i = 1; i < n; i++)
        if (v[i] < v[i1]) i1 = i;
    for (i = 0; i < n; i++)
        if (i != i1 && (i2 == -1 || v[i] < v[i2])) i2 = i;
    for (i = 1; i < n; i++)
        if (v[i] > v[i3]) i3 = i;
    for (i = 0; i < n; i++)
        if (i != i3 && (i4 == -1 || v[i] > v[i4])) i4 = i;

    long long bestP = 0, bestS = 0;
    int first = 1;
    int pairs[3][2];
    pairs[0][0] = i1; pairs[0][1] = i2;
    pairs[1][0] = i1; pairs[1][1] = i3;
    pairs[2][0] = i4; pairs[2][1] = i3;
    for (i = 0; i < 3; i++) {
        int a = pairs[i][0], b = pairs[i][1];
        if (a == b) continue;
        long long prod = v[a] * v[b];
        if (first || prod < bestP) {
            bestP = prod;
            bestS = v[a] + v[b];
            first = 0;
        }
    }
    printf("%lld\n", bestS);
    return 0;
}
