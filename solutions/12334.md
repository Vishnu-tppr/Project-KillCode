# Id 12334 — Sum of Non-prime Digits

```c
#include <stdio.h>

int main() {
    long long N;
    scanf("%lld", &N);
    int sum = 0;
    int found = 0;
    while (N > 0) {
        int d = (int)(N % 10);
        if (d != 2 && d != 3 && d != 5 && d != 7) {
            sum += d;
            found = 1;
        }
        N /= 10;
    }
    if (found)
        printf("%d\n", sum);
    else
        printf("-1\n");
    return 0;
}
```
