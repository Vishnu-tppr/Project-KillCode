# Id 2568 — Fibonacci Sequence

```c
#include <stdio.h>
int main() {
    int n;
    scanf("%d", &n);
    long long a = 0, b = 1;
    for (int i = 0; i < n; i++) {
        if (i) printf(" ");
        printf("%lld", a);
        long long t = a + b;
        a = b;
        b = t;
    }
    printf("\n");
    return 0;
}
```
