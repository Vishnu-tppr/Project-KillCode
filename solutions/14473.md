#include <string.h>

    for (i = 0; i < N; i++) {
        char c;
        scanf(" %c", &c);
        colors[i] = c;
        cnt[(unsigned char)c]++;
    }
    for (i = 0; i < N; i++) {
        if (cnt[(unsigned char)colors[i]] % 2 == 1) {
            printf("%c\n", colors[i]);
            return 0;
        }
    }

}
