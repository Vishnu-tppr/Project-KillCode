# Id 14496 — Sum of Digits - T times

```c
#include <stdio.h>

int main() {
    int N, T;
    scanf("%d", &N);
    scanf("%d", &T);
    int S = 0, x = N;
    if (x == 0) S = 0;
    while (x > 0) { S += x % 10; x /= 10; }
    long long P = (long long)S * T;
    int ds;
    if (P == 0) ds = 0;
    else {
        int r = P % 9;
        ds = (r == 0) ? 9 : r;
    }
    printf("%d\n", ds);
    return 0;
}
```
