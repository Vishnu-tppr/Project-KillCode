# Id 7147 — Matrix - Upper Left to Lower Right

```c
#include <stdio.h>

int main() {
    int n; scanf("%d", &n);
    int a[100][100];
    for (int i = 0; i < n; i++)
        for (int j = 0; j < n - i; j++) scanf("%d", &a[i][j]);
    int b[100][100];
    for (int i = 0; i < n; i++)
        for (int j = 0; j < n - i; j++) {
            b[i][j] = a[i][j];
            b[n - 1 - j][n - 1 - i] = a[i][j];
        }
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            if (j) printf(" ");
            printf("%d", b[i][j]);
        }
        printf("\n");
    }
    return 0;
}
```
