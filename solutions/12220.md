#include <stdio.h>
#include <string.h>
#include <ctype.h>

int main() {
    char s[64];
    int invalid = 0;
    while (fgets(s, sizeof(s), stdin)) {
        int len = (int) strlen(s);
        while (len > 0 && (s[len - 1] == '\n' || s[len - 1] == '\r'))
            s[--len] = '\0';
        if (len == 1 && (s[0] == 'q' || s[0] == 'Q'))
            break;
        int digits = 1;
        int i;
        for (i = 0; i < len; i++)
            if (!isdigit((unsigned char) s[i])) { digits = 0; break; }
        if (!(digits && len == 10))
            invalid++;
    }
    printf("%d\n", invalid);
    return 0;
}
