# Id 14363 — Calculate Player's Score

```c
#include <stdio.h>

int main() {
    long long N;
    scanf("%lld", &N);
    long long sign = (N < 0) ? -1 : 1;
    long long absN = (N < 0) ? -N : N;
    long long rev = 0;
    long long t = absN;
    while (t > 0) {
        rev = rev * 10 + (t % 10);
        t /= 10;
    }
    rev = sign * rev;
    printf("%lld\n", N - rev);
    return 0;
}
```
