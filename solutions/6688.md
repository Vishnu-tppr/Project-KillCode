int calculateMatrixSum(int row, int column, int matrix[row][column]) {
    int sum = 0;
    int i;
    int limit = row < column ? row : column;
    for (i = 0; i < limit; i++) {
        if (matrix[i][i] % 2 != 0) {
            sum += matrix[i][i];
        }
    }

}
