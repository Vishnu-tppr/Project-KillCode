# Id 14482 — function countFascinatingNumbers

long long isqrtll(long long x) {
    if (x < 0) return 0;
    long long lo = 0, hi = x > 1000000000 ? 1000000 : x + 1;
    while (lo < hi) {
        long long mid = (lo + hi + 1) / 2;
        if (mid * mid <= x) lo = mid;
        else hi = mid - 1;
    }
    return lo;
}

int countFascinatingNumbers(int start, int end) {
    long long r1 = isqrtll(end);
    long long l0 = isqrtll(start - 1);
    return (int)(r1 - l0);
}
