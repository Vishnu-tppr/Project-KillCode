# Id 14166 — Camel Case to Normal Sentence

#include <stdio.h>
#include <string.h>

int main() {
    char s[501];
    scanf("%s", s);
    int n = (int)strlen(s);
    int i;
    for (i = 0; i < n; i++) {
        if (s[i] >= 'A' && s[i] <= 'Z') {
            if (i > 0)
                printf(" ");
            printf("%c", s[i] + 32);
        } else {
            printf("%c", s[i]);
        }
    }
    printf("\n");
    return 0;
}
