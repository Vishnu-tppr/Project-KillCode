int magicalArray(int N, int *arr, int K) {
    long long total = 0;
    long long prev = arr[0];
    int i;
    for (i = 1; i < N; i++) {
        long long x = arr[i];
        long long t = 0;
        if (x <= prev)
            t = (prev - x) / K + 1;
        total += t;
        prev = x + t * K;
    }
    return (int) total;
}
