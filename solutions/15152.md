# Id 15152 — Insert Characters - Make Even Length

```c
#include <stdio.h>
#include <string.h>

int main() {
    char s[10005];
    scanf("%s", s);
    int K;
    scanf("%d", &K);
    int L = strlen(s);
    int append = 1;
    char mid[30];
    for (int i = 1; i <= K; i++) {
        if (L % 2 == 1) {
            if (append) {
                s[L] = '*';
                s[L + 1] = 0;
                L++;
            } else {
                for (int k = L; k > 0; k--) s[k] = s[k - 1];
                s[0] = '*';
                L++;
            }
            append = !append;
        }
        sprintf(mid, "#%d#", i);
        int ml = strlen(mid);
        int idx = L / 2;
        for (int k = L - 1; k >= idx; k--) s[k + ml] = s[k];
        for (int k = 0; k < ml; k++) s[idx + k] = mid[k];
        L += ml;
    }
    s[L] = 0;
    printf("%s\n", s);
    return 0;
}
```
