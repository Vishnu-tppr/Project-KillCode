# Id 6822 — Toggle Characters at X

```c
#include <stdio.h>
#include <ctype.h>
int main(){
    char s[1000]; int x,i;
    scanf("%[^ ]%*c%d",s,&x);
    for(i=1;s[i-1]!='\0';i++){
        if(i%x==0){
            char ch=s[i-1];
            if(islower(ch))s[i-1]=toupper(ch);
            else s[i-1]=tolower(ch);
        }
    }
    printf("%s\n",s);
    return 0;
}
```
