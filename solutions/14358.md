int main() {
    int R, C, X, Y;
    long long a[51][51];
    scanf("%d%d", &R, &C);
    for (int i = 0; i < R; i++)
        for (int j = 0; j < C; j++)
            scanf("%lld", &a[i][j]);
    scanf("%d%d", &X, &Y);
    long long tot = 0;
    for (int j = 0; j < C; j++) tot += a[X][j];
    for (int i = 0; i < R; i++) tot += a[i][Y];
    tot -= a[X][Y];
    printf("%lld\n", tot);
    return 0;
}
