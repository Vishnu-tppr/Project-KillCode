```c
int findOddEvenDifference(int SIZE, int *arr) {
    int odd = 0, even = 0;
    for (int i = 0; i < SIZE; i++) {
        if (arr[i] % 2 == 0) {
            even += arr[i];
        } else {
            odd += arr[i];
        }
    }
    return odd - even;
}
```
