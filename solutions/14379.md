# Id 14379 — Game with Multiplication

```c
#include <stdio.h>

int main() {
    long long X, Y, P;
    scanf("%lld", &X);
    scanf("%lld", &Y);
    scanf("%lld", &P);
    long long a = (P + 1) / 2;
    long long b = P / 2;
    long long d = a - b;
    long long A = X * (d ? 2 : 1);
    long long ratio;
    if (A >= Y) ratio = A / Y;
    else ratio = Y / A;
    printf("%lld\n", ratio);
    return 0;
}
```
