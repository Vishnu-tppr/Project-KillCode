#include <string.h>

int main() {
    char s[101];
    scanf("%100s", s);
    int n = strlen(s);
    int cnt[128] = {0};
    for (int i = 0; i < n; i++) cnt[(unsigned char)s[i]]++;
    for (int i = 0; i < n; i++)

}
