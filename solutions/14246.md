#include <stdio.h>

int main() {
    int a[3];
    scanf("%d%d%d", &a[0], &a[1], &a[2]);
    int p = 1000;
    for (int pos = 0; pos < 4; pos++) {
        int mx = 0;
        for (int i = 0; i < 3; i++) {
            int dig = (a[i] / p) % 10;
            if (dig > mx) mx = dig;
        }
        printf("%d", mx);
        p /= 10;
    }
    printf("\n");
    return 0;
}
