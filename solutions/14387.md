# Id 14387 — Multiples with Happy and Coding

```c
#include <stdio.h>

int main() {
    int N, M1, M2;
    scanf("%d %d %d", &N, &M1, &M2);
    int c1 = 0, c2 = 0, c3 = 0;
    int i;
    for (i = 1; i <= N; i++) {
        int a = (i % M1 == 0), b = (i % M2 == 0);
        if (i > 1) printf(" ");
        if (a && b) { printf("HappyCoding"); c3++; }
        else if (a) { printf("Happy"); c1++; }
        else if (b) { printf("Coding"); c2++; }
        else printf("%d", i);
    }
    printf("\nHappy-%d\nCoding-%d\nHappyCoding-%d\n", c1, c2, c3);
    return 0;
}
```
