int calculateTotalTax(int B, int *billingAmounts) {
    long long total = 0;
    int i;
    for (i = 0; i < B; i++) {
        int amount = billingAmounts[i];
        if (amount > 1000) {
            total += (amount - 1000) / 10;
        }
    }
    return (int)total;
}
