#include <stdlib.h>

int *distributeChocolates(int N, int K) {
    int *res = (int *)calloc(K, sizeof(int));
    long long give = 1, remaining = N;
    int i = 0;
    while (remaining > 0) {
        if (remaining >= give) {
            res[i] += (int)give;
            remaining -= give;
        } else {
            res[i] += (int)remaining;
            remaining = 0;
        }
        give++;
        i++;
        if (i == K) i = 0;
    }
    return res;
}
