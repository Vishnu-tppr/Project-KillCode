# Id 14475 — Floors - Even Frequency

#include <stdio.h>

int main() {
    int N;
    scanf("%d", &N);
    int f[1001] = {0};
    for (int i = 0; i < N; i++) {
        int v;
        scanf("%d", &v);
        f[v]++;
    }
    int found = 0;
    for (int v = 1; v <= 1000; v++) {
        if (f[v] > 0 && f[v] % 2 == 0) {
            printf("%d\n", v);
            found = 1;
            break;
        }
    }
    if (!found) printf("-1\n");
    return 0;
}
