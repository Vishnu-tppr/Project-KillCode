# Id 14484 — Parking Lot - Most Filled Row

```c
#include <stdio.h>

int main() {
    int R, C;
    scanf("%d%d", &R, &C);
    int best = -1, bestRow = 1;
    for (int i = 1; i <= R; i++) {
        int cnt = 0;
        for (int j = 0; j < C; j++) {
            int v;
            scanf("%d", &v);
            cnt += v;
        }
        if (cnt > best) { best = cnt; bestRow = i; }
    }
    printf("%d\n", bestRow);
    return 0;
}
```
