# Id 12332 — Swap Every Two Digits

```c
#include <stdio.h>
#include <string.h>
#include <stdlib.h>

static long long swapfy(char *s) {
    int n = (int) strlen(s);
    int i;
    for (i = 0; i + 1 < n; i += 2) {
        char c = s[i];
        s[i] = s[i + 1];
        s[i + 1] = c;
    }
    return strtoll(s, NULL, 10);
}

int main() {
    char x[24], y[24];
    scanf("%s %s", x, y);
    printf("%lld\n", swapfy(x) + swapfy(y));
    return 0;
}
```
