#include <stdio.h>
#include <string.h>
#include <stdlib.h>

int main() {
    char toys[20][30] = {
        "Teddy-Bear", "Lion", "Cat", "Tiger", "Big Teddy-Bear", "Dog",
        "Bunny-Rabbit", "Crocodile", "Husky-Puppy", "Monkey", "Octopus",
        "Kung-Fu-Panda", "Penguin", "Unicorn", "Elephant", "Pink Teddy-Bear",
        "Pikachu", "Doremon", "Hippo", "Tortoise"
    };
    char line[60];
    if (!fgets(line, sizeof line, stdin)) return 0;
    line[strcspn(line, "\n")] = 0;
    int rem = -1;
    if (line[0] >= '0' && line[0] <= '9') {
        int pos = atoi(line);
        if (pos >= 1 && pos <= 20) rem = pos - 1;
    } else {
        for (int i = 0; i < 20; i++)
            if (strcmp(toys[i], line) == 0) { rem = i; break; }
    }
    if (rem == -1) {
        printf("TOY UNAVAILABLE\n");
        return 0;
    }
    int cnt = 1;
    for (int i = 0; i < 20; i++) {
        if (i == rem) continue;
        printf("%d. %s\n", cnt++, toys[i]);
    }
    return 0;
}
