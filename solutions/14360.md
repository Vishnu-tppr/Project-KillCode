# Id 14360 — Validate Coupon Code

```c
#include <stdio.h>
#include <string.h>

int main() {
    char s[24];
    scanf("%s", s);
    int seen[10] = {0};
    int distinct = 0, allsame = 1, i;
    for (i = 0; s[i]; i++) {
        int d = s[i] - '0';
        if (d < 0 || d > 9) continue;
        if (!seen[d]) {
            seen[d] = 1;
            distinct++;
        }
        if (s[i] != s[0]) allsame = 0;
    }
    if (allsame)
        printf("True\n%c\n", s[0]);
    else
        printf("False\n%d\n", distinct);
    return 0;
}
```
