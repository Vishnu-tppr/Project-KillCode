# Id 12254 — Cricket Tournament

```c
#include <stdio.h>
#include <string.h>

int main() {
    char teams[16][64];
    int n = 0;
    char temp[64];
    while (scanf("%s", temp) == 1) {
        if ((temp[0] == 'q' || temp[0] == 'Q') && strlen(temp) == 1)
            break;
        if (n < 16)
            strcpy(teams[n++], temp);
    }
    printf("Total Matches: %d\n", n * (n - 1) / 2);
    int i, j;
    for (i = 0; i < n; i++)
        for (j = i + 1; j < n; j++)
            printf("%s vs %s\n", teams[i], teams[j]);
    return 0;
}
```
