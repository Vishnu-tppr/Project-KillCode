#include <stdio.h>

int main() {
    char s[101];
    scanf("%s", s);
    int star = 0, hash = 0, i;
    for (i = 0; s[i]; i++) {
        if (s[i] == '*') star++;
        else if (s[i] == '#') hash++;
    }
    printf("%d\n", star - hash);
    return 0;
}
