# Id 7076 — Alphabet Integer Pattern

```c
#include <stdio.h>
int main(){
    char c1,c2; int x,y;
    scanf(" %c %d",&c1,&x);
    scanf(" %c %d",&c2,&y);
    int first=1;
    for(char ch=c1;ch<=c2;ch++){
        int start, end;
        if(ch==c1){ start=x; end=10; }
        else if(ch==c2){ start=1; end=y; }
        else { start=1; end=10; }
        for(int v=start;v<=end;v++){
            if(!first) printf(" ");
            printf("%c%d",ch,v);
            first=0;
        }
    }
    return 0;
}
```
