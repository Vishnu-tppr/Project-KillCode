# Id 12373 — Count Submatrices - Product K

```c
#include <stdio.h>

int main() {
    int R, C, i, j;
    scanf("%d %d", &R, &C);
    int m[10][10];
    for (i = 0; i < R; i++)
        for (j = 0; j < C; j++)
            scanf("%d", &m[i][j]);
    long long K;
    scanf("%lld", &K);
    int v00 = m[0][0];
    int count = 0;
    int r1, c1, r2, c2, a, b;
    for (r1 = 0; r1 < R; r1++) {
        for (c1 = 0; c1 < C; c1++) {
            for (r2 = r1 + 1; r2 < R; r2++) {
                for (c2 = c1 + 1; c2 < C; c2++) {
                    long long p = 1;
                    int has = 0;
                    for (a = r1; a <= r2 && p <= K; a++) {
                        for (b = c1; b <= c2; b++) {
                            if (m[a][b] == v00) has = 1;
                            p *= m[a][b];
                            if (p > K) break;
                        }
                    }
                    if (p <= K && has) count++;
                }
            }
        }
    }
    printf("%d\n", count);
    return 0;
}
```
