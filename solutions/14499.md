#include <stdio.h>
#include <string.h>

int main() {
    char s[1001];
    int N;
    scanf("%s %d", s, &N);
    int len = strlen(s);
    int mx = 0;
    int i, j;
    for (i = 0; i < len; i += N) {
        int cnt = 0;
        for (j = i; j < len && j < i + N; j++)
            if (s[j] == 'b') cnt++;
        if (cnt > mx) mx = cnt;
    }
    printf("%d\n", mx);
    return 0;
}
