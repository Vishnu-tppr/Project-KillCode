# Id 2584 — Reverse String Till Underscore

```c
#include <stdio.h>
#include <string.h>

int main() {
    char s[101];
    scanf("%100s", s);
    int n = strlen(s);
    int u = -1;
    for (int i = 0; i < n; i++) if (s[i] == '_') { u = i; break; }
    int end = (u == -1) ? n : u;
    for (int i = 0, j = end - 1; i < j; i++, j--) {
        char t = s[i]; s[i] = s[j]; s[j] = t;
    }
    printf("%s\n", s);
    return 0;
}
```
