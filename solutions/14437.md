int countCommonCharacters(char *str1, char *str2)
{
    int seen1[256] = {0}, seen2[256] = {0};
    int i;
    for (i = 0; str1[i]; i++)
    {
        char c = str1[i];
        if (c >= 'A' && c <= 'Z') c += 'a' - 'A';
        seen1[(unsigned char)c] = 1;
    }
    for (i = 0; str2[i]; i++)
    {
        char c = str2[i];
        if (c >= 'A' && c <= 'Z') c += 'a' - 'A';
        seen2[(unsigned char)c] = 1;
    }
    int cnt = 0;
    for (i = 0; i < 256; i++)
        if (seen1[i] && seen2[i])
            cnt++;
    return cnt;
}
