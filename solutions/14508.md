#include <stdio.h>

int main() {
    int N, T, X, Y;
    scanf("%d %d", &N, &T);
    scanf("%d %d", &X, &Y);
    long long num = (long long)N - (long long)Y * T;
    long long den = (long long)X - Y;
    if (den == 0 || num % den != 0) {
        printf("-1\n");
        return 0;
    }
    long long p = num / den;
    if (p < 0 || p > T) printf("-1\n");
    else printf("%lld\n", p);
    return 0;
}
