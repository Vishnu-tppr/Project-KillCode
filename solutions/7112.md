# Id 7112 — Multiply with the Minimum Adjacent

#include <stdio.h>

int main() {
    int n;
    scanf("%d", &n);
    int a[100];
    for (int i = 0; i < n; i++) scanf("%d", &a[i]);
    for (int i = 0; i < n; i++) {
        long long p;
        if (i == 0) p = (long long)a[i] * a[i+1];
        else if (i == n-1) p = (long long)a[i] * a[i-1];
        else {
            int m = a[i-1] < a[i+1] ? a[i-1] : a[i+1];
            p = (long long)a[i] * m;
        }
        if (i) printf(" ");
        printf("%lld", p);
    }
    printf("\n");
    return 0;
}
