# Id 7088 — Palindromic Integers

```c
#include <stdio.h>

int isPal(long long n) {
    long long rev = 0, t = n;
    while (t) { rev = rev * 10 + t % 10; t /= 10; }
    return rev == n;
}

int main() {
    int n, f = 0;
    scanf("%d", &n);
    for (int i = 0; i < n; i++) {
        long long x;
        scanf("%lld", &x);
        if (isPal(x)) { printf("%lld\n", x); f = 1; }
    }
    if (!f) printf("-1\n");
    return 0;
}
```
