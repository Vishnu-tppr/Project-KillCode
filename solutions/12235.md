# Id 12235 — Chocolate Vending Machine

```c
#include <stdio.h>

int main() {
    int N, K, C, i;
    scanf("%d %d", &N, &K);
    scanf("%d", &C);
    int avail = N;
    for (i = 1; i <= C; i++) {
        int req;
        scanf("%d", &req);
        printf("Customer: %d\n", i);
        if (req > avail) {
            printf("Invalid\n");
        } else {
            avail -= req;
            if (avail <= K) avail = N;
            printf("Number of chocolates sold: %d\n", req);
            printf("Number of chocolates available: %d\n", avail);
        }
    }
    return 0;
}
```
