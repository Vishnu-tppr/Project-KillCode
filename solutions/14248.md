char *bitwise(int input1)
{
    static char buf[40];
    unsigned int n = (unsigned int)input1;
    int set = 0;
    unsigned int x = n;
    while (x) { set += x & 1; x >>= 1; }
    if (set == 0)
    {
        sprintf(buf, "0#0#0");
        return buf;
    }
    int lo = 0, hi = 0, i;
    for (i = 0; i < 32; i++)
        if (n & (1u << i)) { lo = i; break; }
    for (i = 31; i >= 0; i--)
        if (n & (1u << i)) { hi = i; break; }
    sprintf(buf, "%d#%d#%d", set, 32 - lo, 32 - hi);
    return buf;
}
