# Id 15151 — Sum - Count of Subsequences

```c
#include <stdio.h>

int main() {
    int N;
    scanf("%d", &N);
    long long cnt = 0;
    for (int i = 0; i < N; i++) {
        long long v;
        scanf("%lld", &v);
        if (v == 1) cnt++;
    }
    printf("%lld\n", cnt);
    return 0;
}
```
