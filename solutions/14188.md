# Id 14188 — Reverse String - Preserve Digits

```c
#include <stdio.h>
#include <string.h>
#include <ctype.h>

int main() {
    char s[101];
    char Ls[101];
    scanf("%s", s);
    int L = strlen(s);
    int n = 0;
    for (int i = 0; i < L; i++)
        if (!isdigit((unsigned char)s[i])) Ls[n++] = s[i];
    for (int i = 0; i < L; i++) {
        if (!isdigit((unsigned char)s[i])) {
            s[i] = Ls[--n];
        }
    }
    printf("%s\n", s);
    return 0;
}
```
