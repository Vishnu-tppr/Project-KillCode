```c
int removeIntegersRecursively(int SIZE, int *arr) {
    while (SIZE > 1) {
        int k = 0;
        for (int i = 1; i < SIZE; i += 2) {
            arr[k++] = arr[i];
        }
        SIZE = k;
    }
    return arr[0];
}
```
