# Id 12270 — Most Favorite Place in India

```c
#include <stdio.h>

int main() {
    const char *names[] = {"", "Jaipur", "Agra", "Rishikesh", "Goa", "Mysore"};
    int count[6] = {0};
    int v;
    int max = 0;
    while (scanf("%d", &v) == 1) {
        if (v >= 1 && v <= 5) {
            count[v]++;
            if (count[v] > max) max = count[v];
        }
    }
    int i;
    for (i = 1; i <= 5; i++)
        if (count[i] == max)
            printf("%s\n", names[i]);
    return 0;
}
```
