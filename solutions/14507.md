#include <string.h>

int main() {
    char s[1001];
    scanf("%s", s);
    int cnt[26] = {0}, i;
    for (i = 0; s[i]; i++) {
        char c = s[i];
        if (c >= 'A' && c <= 'Z') c = c - 'A' + 'a';
        cnt[c - 'a']++;
    }
    for (i = 0; i < 26; i++) {
        if (cnt[i] % 2) {
            printf("NO\n");
            return 0;
        }
    }
    printf("YES\n");
    return 0;
}
