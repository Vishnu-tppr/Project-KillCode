# Id 7034 — Four Integers - Adjacent

```c
#include <stdio.h>
int main(){
    int a[4];
    for(int i=0;i<4;i++) scanf("%d",&a[i]);
    int ok=1;
    for(int i=0;i<4;i++)
        for(int j=-1;j<=1;j+=2){
            int k=i+j;
            if(k>=0&&k<4 && (a[i]%2)==(a[k]%2)) ok=0;
        }
    printf(ok?"YES":"NO");
    return 0;
}
```
