# Id 6929 — Sum of N Integers Except Current Integer

```c
#include <stdio.h>
int main(){
    int n; int a[100];
    long long total=0; int i;
    scanf("%d",&n);
    for(i=0;i<n;i++){scanf("%d",&a[i]);total+=a[i];}
    for(i=0;i<n;i++){if(i>0)printf(" ");printf("%lld",total-a[i]);}
    printf("\n");
    return 0;
}
```
