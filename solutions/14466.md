int countDistinctYears(char *str) {
    int years[100];
    int cnt = 0;
    int i = 0;
    while (str[i] != '\0') {
        if (str[i] >= '0' && str[i] <= '9' &&
            str[i + 1] >= '0' && str[i + 1] <= '9' &&
            str[i + 2] == '-' &&
            str[i + 3] >= '0' && str[i + 3] <= '9' &&
            str[i + 4] >= '0' && str[i + 4] <= '9' &&
            str[i + 5] == '-' &&
            str[i + 6] >= '0' && str[i + 6] <= '9' &&
            str[i + 7] >= '0' && str[i + 7] <= '9' &&
            str[i + 8] >= '0' && str[i + 8] <= '9' &&
            str[i + 9] >= '0' && str[i + 9] <= '9') {
            int y = (str[i + 6] - '0') * 1000 + (str[i + 7] - '0') * 100 +
                    (str[i + 8] - '0') * 10 + (str[i + 9] - '0');
            int dup = 0;
            for (int j = 0; j < cnt; j++) {
                if (years[j] == y) { dup = 1; break; }
            }
            if (!dup) years[cnt++] = y;
            i += 10;
        } else {
            i++;
        }
    }
    return cnt;
}
