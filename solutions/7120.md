# Id 7120 — Space(s) after Punctuation Mark(s)

```c
#include <stdio.h>
#include <string.h>

int isPunc(char c) { return c == '.' || c == ',' || c == '!' || c == '?'; }

int main() {
    char s[101];
    fgets(s, sizeof(s), stdin);
    s[strcspn(s, "\n")] = 0;
    for (int i = 0; s[i]; i++) {
        printf("%c", s[i]);
        if (isPunc(s[i]) && s[i+1] && s[i+1] != ' ')
            printf(" ");
    }
    printf("\n");
    return 0;
}
```
