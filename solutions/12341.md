    int M, N;

    int a[51][51];
    for (int i = 0; i < M; i++)
        for (int j = 0; j < N; j++)
            scanf("%d", &a[i][j]);
    for (int j = 0; j < N; j++) {
        int mn = a[0][j];
        for (int i = 1; i < M; i++)
            if (a[i][j] < mn) mn = a[i][j];

}
