# Id 14530 — Triangle and Inverted Triangle Pattern

```c
#include <stdio.h>

int main() {
    int N;
    scanf("%d", &N);
    int i, j, k;
    for (i = 1; i <= N; i++) {
        for (k = 1; k < i; k++) printf("*");
        for (j = i; j <= N; j++) {
            if (j > i) printf(" ");
            printf("%d", j);
        }
        printf("\n");
    }
    for (i = N - 1; i >= 1; i--) {
        for (k = 1; k < i; k++) printf("*");
        for (j = i; j <= N; j++) {
            if (j > i) printf(" ");
            printf("%d", j);
        }
        printf("\n");
    }
    return 0;
}
```
