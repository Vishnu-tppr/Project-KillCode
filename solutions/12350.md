int findMaxDifference(int SIZE, int *arr)
{
    int maxDiff = arr[0] - arr[1];
    int i;
    for (i = 1; i + 1 < SIZE; i++) {
        int diff = arr[i] - arr[i + 1];
        if (diff > maxDiff)
            maxDiff = diff;
    }
    return maxDiff;
}
