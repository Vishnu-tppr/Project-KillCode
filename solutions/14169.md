# Id 14169 — Find Missing Alphabets

```c
#include <stdio.h>

int main() {
    int pres[26] = {0};
    int c;
    while ((c = getchar()) != EOF && c != '\n') {
        if (c >= 'A' && c <= 'Z') pres[c - 'A'] = 1;
        else if (c >= 'a' && c <= 'z') pres[c - 'a'] = 1;
    }
    int found = 0;
    for (int i = 0; i < 26; i++) {
        if (!pres[i]) {
            putchar('a' + i);
            found = 1;
        }
    }
    if (!found) printf("0");
    printf("\n");
    return 0;
}
```
