#include <stdio.h>

int main() {
    const char *coffee[] = {"Espresso Coffee", "Cappuccino Coffee", "Latte Coffee"};
    const char *tea[] = {"Plain Tea", "Assam Tea", "Ginger Tea", "Cardamom Tea",
                         "Masala Tea", "Lemon Tea", "Green Tea", "Organic Darjeeling Tea"};
    const char *soup[] = {"Hot and Sour Soup", "Veg Corn Soup", "Tomato Soup", "Spicy Tomato Soup"};
    const char *bev[] = {"Hot Chocolate Drink", "Badam Drink", "Badam-Pista Drink"};

    char m;
    int s;
    scanf("%c", &m);
    scanf("%d", &s);

    const char **items = NULL;
    int count = 0;
    if (m == 'C' || m == 'c') { items = coffee; count = 3; }
    else if (m == 'T' || m == 't') { items = tea; count = 8; }
    else if (m == 'S' || m == 's') { items = soup; count = 4; }
    else if (m == 'B' || m == 'b') { items = bev; count = 3; }

    if (items != NULL && s >= 1 && s <= count)
        printf("Enjoy Your %s\n", items[s - 1]);
    else
        printf("Invalid Option!\n");
    return 0;
}
