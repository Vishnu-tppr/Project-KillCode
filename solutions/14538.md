# Id 14538 — Halindrome (Half Palindrome)

```c
#include <stdio.h>
#include <string.h>

char s[101];

int isPal(int start, int len) {
    int i;
    for (i = 0; i < len / 2; i++) {
        if (s[start + i] != s[start + len - 1 - i]) return 0;
    }
    return 1;
}

int halindrome(int start, int len) {
    if (len >= 2 && isPal(start, len)) return 1;
    if (len < 2) return 0;
    int half = len / 2;
    if (halindrome(start, half)) return 1;
    if (halindrome(start + len - half, half)) return 1;
    return 0;
}

int main() {
    scanf("%s", s);
    printf("%s\n", halindrome(0, strlen(s)) ? "YES" : "NO");
    return 0;
}
```
