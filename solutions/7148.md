#include <stdio.h>

int main(){
    int r, c; scanf("%d %d", &r, &c);
    int m[100][100], rz[100] = {0}, cz[100] = {0};
    for (int i = 0; i < r; i++)
        for (int j = 0; j < c; j++) {
            scanf("%d", &m[i][j]);
            if (m[i][j] == 0) { rz[i] = 1; cz[j] = 1; }
        }
    for (int i = 0; i < r; i++) {
        for (int j = 0; j < c; j++) {
            if (j) printf(" ");
            printf("%d", (rz[i] || cz[j]) ? 0 : m[i][j]);
        }
        printf("\n");
    }
    return 0;
}
