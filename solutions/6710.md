# Id 6710 — Print All Consonants

```c
#include <stdio.h>

int isvowel(char c) {
    return c == 'a' || c == 'e' || c == 'i' || c == 'o' || c == 'u';
}

int main() {
    char ch;
    scanf("%c", &ch);
    for (char c = ch; c <= 'z'; c++)
        if (!isvowel(c)) printf("%c", c);
    printf("\n");
    return 0;
}
```
