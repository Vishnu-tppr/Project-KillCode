# Id 14369 — Efficient Carry Weight Bags

```c
int efficientCarry(int N, float *weight) {
    int w[1000];
    for (int i = 0; i < N; i++) {
        w[i] = (int)(weight[i] * 100 + 0.5);
    }
    for (int i = 0; i < N; i++) {
        for (int j = i + 1; j < N; j++) {
            if (w[j] < w[i]) {
                int t = w[i]; w[i] = w[j]; w[j] = t;
            }
        }
    }
    int trips = 0;
    int i = 0, j = N - 1;
    while (i <= j) {
        if (i < j && w[i] + w[j] <= 300) {
            i++;
        }
        j--;
        trips++;
    }
    return trips;
}
```
