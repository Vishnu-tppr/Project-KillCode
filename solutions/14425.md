#include <stdio.h>
#include <string.h>

int main() {
    char p[25];
    char ch[3];
    char children[50][25];
    int nc = 0;
    scanf("%s", p);
    scanf("%s", ch);
    if (ch[0] == 'Y' || ch[0] == 'y') {
        char line[2000];
        scanf(" %[^\n]", line);
        char *tok = strtok(line, ",");
        while (tok) {
            strcpy(children[nc++], tok);
            tok = strtok(NULL, ",");
        }
    }
    printf("TOTAL MEMBERS: %d\n", nc + 1);
    printf("COMMISSION DETAILS\n");
    int pc = (nc == 0) ? 150 : nc * 300;
    printf("%s: %d INR\n", p, pc);
    for (int i = 0; i < nc; i++)
        printf("%s: 150 INR\n", children[i]);
    return 0;
}
