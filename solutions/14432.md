# Id 14432 — Reverse Binary in 32-Bit

```c
#include <stdio.h>

static unsigned int rev32(unsigned int x) {
    unsigned int r = 0;
    int i;
    for (i = 0; i < 32; i++) {
        r = (r << 1) | (x & 1u);
        x >>= 1;
    }
    return r;
}

int main() {
    unsigned int x;
    int first = 1;
    while (scanf("%u", &x) == 1) {
        if (!first) printf(", ");
        first = 0;
        printf("%u", rev32(x));
    }
    printf("\n");
    return 0;
}
```
