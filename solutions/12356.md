# Id 12356 — Area of Largest House

```c
#include <stdio.h>

int grid[50][50], R, C;

int dfs(int r, int c) {
    if (r < 0 || r >= R || c < 0 || c >= C || !grid[r][c]) return 0;
    grid[r][c] = 0;
    return 1 + dfs(r + 1, c) + dfs(r - 1, c) + dfs(r, c + 1) + dfs(r, c - 1);
}

int main() {
    scanf("%d %d", &R, &C);
    int i, j;
    for (i = 0; i < R; i++)
        for (j = 0; j < C; j++)
            scanf("%d", &grid[i][j]);
    int mx = 0;
    for (i = 0; i < R; i++)
        for (j = 0; j < C; j++)
            if (grid[i][j]) {
                int a = dfs(i, j);
                if (a > mx) mx = a;
            }
    printf("%d\n", mx);
    return 0;
}
```
