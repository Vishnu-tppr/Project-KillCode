#include <stdio.h>

int main() {
    int N, K;
    scanf("%d", &N);
    int arr[100];
    int i;
    for (i = 0; i < N; i++)
        scanf("%d", &arr[i]);
    scanf("%d", &K);
    for (i = K; i < N; i++) {
        if (i > K)
            printf(" ");
        printf("%d", arr[i]);
    }
    for (i = 0; i < K; i++) {

    }
    printf("\n");

}
