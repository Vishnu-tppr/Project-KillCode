# Id 6917 — Negative Integers in Reverse Order

```c
#include <stdio.h>
int main(){
    int n; int a[50];
    scanf("%d",&n);
    int i,neg[50],cnt=0;
    for(i=0;i<n;i++){scanf("%d",&a[i]);
        if(a[i]<0)neg[cnt++]=a[i];
    }
    if(cnt==0)printf("-1\n");
    else{
        for(i=cnt-1;i>=0;i--){if(i<cnt-1)printf(" ");printf("%d",neg[i]);}
        printf("\n");
    }
    return 0;
}
```
