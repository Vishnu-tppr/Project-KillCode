# Id 14440 — ASCII Shift Encryption

```c
#include <stdio.h>
#include <string.h>

int main() {
    char s[100005];
    scanf("%s", s);
    int i, len = strlen(s);
    for (i = 0; i < len; i++) {
        int shift = (int)s[i] % 26;
        s[i] = (char)('a' + (s[i] - 'a' + shift) % 26);
    }
    printf("%s\n", s);
    return 0;
}
```
