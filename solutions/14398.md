#include <stdio.h>

int main() {
    int nums[3];
    scanf("%d %d %d", &nums[0], &nums[1], &nums[2]);
    int sumL = 0, sumS = 0;
    int k;
    for (k = 0; k < 3; k++) {
        int cnt[10] = {0};
        int n = nums[k];
        while (n > 0) {
            cnt[n % 10]++;
            n /= 10;
        }
        int big = -1, d;
        for (d = 9; d >= 0; d--)
            if (cnt[d]) { big = d; break; }
        int second;
        if (cnt[big] >= 2)
            second = big;
        else {
            second = -1;
            for (d = big - 1; d >= 0; d--)
                if (cnt[d]) { second = d; break; }
        }
        sumL += big;
        sumS += second;
    }
    printf("%d\n", sumL - sumS);
    return 0;
}
