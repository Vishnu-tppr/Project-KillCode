# Id 14376 — Longest Substring Length - K Distinct Vowels

```c
#include <stdio.h>
#include <string.h>

int main() {
    char s[1001];
    int K;
    scanf("%s", s);
    scanf("%d", &K);
    int n = (int)strlen(s);
    int cnt[5] = {0};
    int l = 0;
    int distinct = 0;
    int best = -1;
    int i;
    for (i = 0; i < n; i++) {
        int v = -1;
        if (s[i] == 'a') v = 0;
        else if (s[i] == 'e') v = 1;
        else if (s[i] == 'i') v = 2;
        else if (s[i] == 'o') v = 3;
        else if (s[i] == 'u') v = 4;
        if (v >= 0) {
            if (cnt[v] == 0) distinct++;
            cnt[v]++;
        }
        while (distinct > K) {
            int lv = -1;
            if (s[l] == 'a') lv = 0;
            else if (s[l] == 'e') lv = 1;
            else if (s[l] == 'i') lv = 2;
            else if (s[l] == 'o') lv = 3;
            else if (s[l] == 'u') lv = 4;
            if (lv >= 0) {
                cnt[lv]--;
                if (cnt[lv] == 0) distinct--;
            }
            l++;
        }
        if (distinct == K && i - l + 1 > best)
            best = i - l + 1;
    }
    printf("%d\n", best);
    return 0;
}
```
