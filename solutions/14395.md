# Id 14395 — Increasing Diagonals Pattern

```c
#include <stdio.h>

int main() {
    int N;
    scanf("%d", &N);
    int a[50][50] = {0};
    int cnt = 1;
    for (int k = N - 1; k <= 2 * (N - 1); k++) {
        for (int r = 0; r < N; r++) {
            int c = k - r;
            if (c >= 0 && c < N) a[r][c] = cnt++;
        }
    }
    for (int r = 0; r < N; r++) {
        for (int c = 0; c < N; c++) {
            if (c) printf(" ");
            if (a[r][c] == 0) printf("*");
            else printf("%d", a[r][c]);
        }
        printf("\n");
    }
    return 0;
}
```
