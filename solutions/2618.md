#define LIMBS 30
typedef struct { int d[LIMBS]; int n; } Big;
void mul(Big *r, long long m) {
    unsigned long long carry = 0;
    for (int i = 0; i < r->n; i++) {
        unsigned long long cur = (unsigned long long)r->d[i] * (unsigned long long)m + carry;
        r->d[i] = (int)(cur % 1000000000ULL);
        carry = cur / 1000000000ULL;
    }
    while (carry) { r->d[r->n++] = (int)(carry % 1000000000ULL); carry /= 1000000000ULL; }
}
void divB(Big *q, const Big *a, long long m) {
    unsigned long long rem = 0;
    for (int i = a->n - 1; i >= 0; i--) {
        unsigned long long cur = rem * 1000000000ULL + (unsigned long long)a->d[i];
        q->d[i] = (int)(cur / (unsigned long long)m);
        rem = cur % (unsigned long long)m;
    }
    q->n = a->n;
    while (q->n > 1 && q->d[q->n - 1] == 0) q->n--;
}
void printB(const Big *b) {
    printf("%d", b->d[b->n - 1]);
    for (int i = b->n - 2; i >= 0; i--) printf("%09d", b->d[i]);
}
