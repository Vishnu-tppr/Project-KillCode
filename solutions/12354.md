# Id 12354 — Maximum Length - 0s or 1s

```c
#include <stdio.h>
#include <string.h>

int main() {
    char s[101];
    scanf("%s", s);
    int L = strlen(s);
    int best = -1;
    int i = 0;
    while (i < L) {
        int j = i;
        while (j + 1 < L && s[j + 1] == s[i]) j++;
        int len = j - i + 1;
        int startRun = (i == 0), endRun = (j == L - 1);
        if (!startRun && !endRun && len > best) best = len;
        i = j + 1;
    }
    printf("%d\n", best);
    return 0;
}
```
