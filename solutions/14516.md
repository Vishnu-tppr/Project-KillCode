# Id 14516 — Collect Maximum Fruits

```c
#include <stdio.h>

int main() {
    int n, i;
    scanf("%d", &n);
    long long a[100];
    for (i = 0; i < n; i++)
        scanf("%lld", &a[i]);
    int x, k;
    scanf("%d %d", &x, &k);
    long long best = -1;
    int s;
    for (s = 0; s + k <= n; s++) {
        if (x - 1 < s || x - 1 >= s + k) continue;
        long long sum = 0;
        for (i = s; i < s + k; i++)
            sum += a[i];
        if (sum > best) best = sum;
    }
    printf("%lld\n", best);
    return 0;
}
```
