# Id 6610 — Arithmetic Operation - Odd or Even

```c
#include <stdio.h>
#include <stdlib.h>

int main() {
    int x, y;
    scanf("%d%d", &x, &y);
    if (x % 2 == 0 && y % 2 == 0)
        printf("%d\n", x + y);
    else if (x % 2 != 0 && y % 2 != 0)
        printf("%d\n", x * y);
    else
        printf("%d\n", abs(x - y));
    return 0;
}
```
