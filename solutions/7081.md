# Id 7081 — Integer with Hyphen Pattern

```c
#include <stdio.h>
int main(){
    int N;
    scanf("%d",&N);
    for(int i=1;i<=N;i++){
        for(int d=1;d<=i;d++) printf("%d",d);
        for(int h=0;h<N-i;h++) printf("-");
        for(int d=i;d>=1;d--) printf("%d",d);
        printf("\n");
    }
    return 0;
}
```
