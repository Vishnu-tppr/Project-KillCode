# Id 7139 — Matrix - Rows Odd/Even

```c
#include <stdio.h>

int main() {
    int r, c;
    scanf("%d %d", &r, &c);
    int m[100][100];
    for (int i = 0; i < r; i++)
        for (int j = 0; j < c; j++)
            scanf("%d", &m[i][j]);
    for (int i = 0; i < r / 2; i++) {
        for (int j = 0; j < c; j++) {
            if ((m[i][j] % 2) != (m[r-1-i][j] % 2)) {
                printf("No\n");
                return 0;
            }
        }
    }
    printf("Yes\n");
    return 0;
}

```
