# Id 2525 — Odd Integers In Range

```c
#include <stdio.h>
int main() {
    int x, y, first = 1;
    scanf("%d %d", &x, &y);
    for (int i = x + 1; i < y; i++) {
        if (i % 2 != 0) {
            if (!first) printf(" ");
            printf("%d", i);
            first = 0;
        }
    }
    printf("\n");
    return 0;
}
```
