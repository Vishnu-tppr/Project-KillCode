# Id 14521 — Break Integer - Maximum Product

```c
#include <stdio.h>

long long f[101];

long long dp(int i) {
    if (f[i] != -1) return f[i];
    if (i == 1) return f[i] = 1;
    long long best = i;
    for (int j = 1; j < i; j++) {
        long long v = j * dp(i - j);
        if (v > best) best = v;
    }
    return f[i] = best;
}

int main() {
    int N;
    scanf("%d", &N);
    for (int i = 0; i < 101; i++) f[i] = -1;
    long long ans = 0;
    for (int j = 1; j < N; j++) {
        long long v = j * dp(N - j);
        if (v > ans) ans = v;
    }
    printf("%lld\n", ans);
    return 0;
}
```
