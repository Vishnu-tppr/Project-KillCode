# Id 14184 — Maximum Sum - Integer with Index

```c
#include <stdio.h>

int main() {
    int N;
    scanf("%d", &N);
    int arr[100];
    int i, j;
    for (i = 0; i < N; i++)
        scanf("%d", &arr[i]);
    for (i = 0; i < N; i++) {
        for (j = i + 1; j < N; j++) {
            if (arr[i] > arr[j]) {
                int t = arr[i];
                arr[i] = arr[j];
                arr[j] = t;
            }
        }
    }
    long long sum = 0;
    for (i = 0; i < N; i++)
        sum += (long long)arr[i] * i;
    printf("%lld\n", sum);
    return 0;
}
```
