    while (n) { digits[nd++] = n % 10; n /= 10; }
    long long sum = 0;
    for (int i = 0; i < nd; i++) {
        long long p = 1;
        for (int k = 0; k < nd; k++) p *= digits[i];
        sum += p;
    }
    int res = 0;
    if (sum == X) {
        for (int i = 0; i < nd; i++) if (digits[i] % 2 == 0) res += digits[i];
    } else {
        for (int i = 0; i < nd; i++) if (digits[i] % 2) res += digits[i];

}
