# Id 14382 — Cyclic Increasing Array or Not

```c
#include <stdio.h>

int main() {
    int N;
    scanf("%d", &N);
    int a[1001];
    for (int i = 0; i < N; i++) scanf("%d", &a[i]);
    int mi = 0;
    for (int i = 1; i < N; i++)
        if (a[i] < a[mi]) mi = i;
    int ok = 1;
    for (int k = 0; k < N - 1; k++) {
        int x = (mi + k) % N;
        int y = (mi + k + 1) % N;
        if (a[x] > a[y]) { ok = 0; break; }
    }
    if (ok) printf("True\n"); else printf("False\n");
    printf("%d\n", mi);
    return 0;
}
```
