int main() {
    static char buf[8192];
    size_t len = fread(buf, 1, sizeof(buf) - 1, stdin);

    int nums[40], nc = 0;
    char *p = buf;
    while (*p) {
        while (*p && !(*p >= '0' && *p <= '9')) p++;
        if (!*p) break;
        nums[nc++] = (int) strtol(p, &p, 10);
    }

    int n = nums[0];
    int ages[10];
    int i;
    for (i = 0; i < n; i++)
        ages[i] = nums[i + 1];
    for (i = 0; i < n; i++)
        if (ages[i] < 10 || ages[i] > 100) {
            printf("INVALID INPUT\n");
            return 0;
        }
    int senior = 0;
    for (i = 0; i < n; i++)
        if (ages[i] > 60) senior++;
    int q2 = senior < 5 ? senior : 5;
    int q1 = n - q2;
    printf("Q1:%d minutes\nQ2:%d minutes\n", q1 * 15, q2 * 15);
    return 0;
}
