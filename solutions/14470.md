# Id 14470 — Copper Wires - Minimum Cost

```c
#include <stdio.h>

int main() {
    int N;
    scanf("%d", &N);
    long long a[31];
    int i;
    for (i = 0; i < N; i++) scanf("%lld", &a[i]);
    long long total = 0;
    int sz = N;
    while (sz > 1) {
        int bi = 0;
        long long bsum = a[0] + a[1];
        for (i = 1; i + 1 < sz; i++) {
            if (a[i] + a[i + 1] < bsum) { bsum = a[i] + a[i + 1]; bi = i; }
        }
        total += bsum;
        a[bi] = bsum;
        for (i = bi + 1; i + 1 < sz; i++) a[i] = a[i + 1];
        sz--;
    }
    printf("%lld\n", total);
    return 0;
}
```
