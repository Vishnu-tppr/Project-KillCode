# Id 7145 — Replace the Common Elements - Matrix

```c
#include <stdio.h>

int main() {
    int n;
    scanf("%d", &n);
    int m[50][50];
    for (int i = 0; i < n; i++)
        for (int j = 0; j < n; j++)
            scanf("%d", &m[i][j]);
    int isCommon[100];
    for (int v = 1; v <= 99; v++) isCommon[v] = 1;
    for (int i = 0; i < n; i++) {
        int rowSeen[100] = {0};
        for (int j = 0; j < n; j++) rowSeen[m[i][j]] = 1;
        for (int v = 1; v <= 99; v++)
            if (!rowSeen[v]) isCommon[v] = 0;
    }
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            if (j) printf(" ");
            printf("%d", isCommon[m[i][j]] ? -1 : m[i][j]);
        }
        printf("\n");
    }
    return 0;
}
```
