# Id 14513 — Integers - Occurrence Count

```c
#include <stdio.h>
#include <string.h>

int main() {
    static int cnt[100001];
    char line[500000];
    int first[5000], n = 0, i;

    fgets(line, sizeof(line), stdin);
    {
        char *tok = strtok(line, ",\n");
        while (tok) {
            sscanf(tok, "%d", &first[n++]);
            tok = strtok(NULL, ",\n");
        }
    }
    fgets(line, sizeof(line), stdin);
    {
        char *tok = strtok(line, ",\n");
        while (tok) {
            int v;
            sscanf(tok, "%d", &v);
            cnt[v]++;
            tok = strtok(NULL, ",\n");
        }
    }
    for (i = 0; i < n; i++) {
        printf("%d-%d\n", first[i], cnt[first[i]]);
    }
    return 0;
}
```
