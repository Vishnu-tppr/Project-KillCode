# Id 14527 — Triangular Number - Factors Sum

```c
#include <stdio.h>

int main() {
    long long N;
    scanf("%lld", &N);
    long long T = N * (N + 1) / 2;
    long long sum = 0;
    for (long long i = 1; i * i <= T; i++) {
        if (T % i == 0) {
            sum += i;
            if (i != T / i) sum += T / i;
        }
    }
    printf("%lld\n", sum - T);
    return 0;
}
```
