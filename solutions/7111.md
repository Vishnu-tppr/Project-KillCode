# Id 7111 — Contiguous Integers or Not

```c
#include <stdio.h>
#include <stdlib.h>

int cmp(const void *a, const void *b) { return *(int*)a - *(int*)b; }

int main() {
    int n;
    scanf("%d", &n);
    int a[100];
    for (int i = 0; i < n; i++) scanf("%d", &a[i]);
    qsort(a, n, sizeof(int), cmp);
    for (int i = 1; i < n; i++)
        if (a[i] - a[i-1] > 1) { printf("NO\n"); return 0; }
    printf("YES\n");
    return 0;
}

```
