#include <string.h>

int main() {
    char s[1001];
    scanf("%s", s);
    int L = strlen(s);
    char c = s[0];
    int last = 0;
    for (int i = 0; i < L; i++)
        if (s[i] == c) last = i;
    printf("%d\n", last - 1);

}
