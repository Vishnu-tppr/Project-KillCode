# Id 14412 — Square Shaped Plots

```c
#include <stdio.h>

int main() {
    int N, v;
    scanf("%d", &N);
    int c = 0;
    for (int i = 0; i < N; i++) {
        scanf("%d", &v);
        int sq = 0;
        for (int r = 1; r * r <= v; r++) {
            if (r * r == v) { sq = 1; break; }
        }
        if (sq) c++;
    }
    printf("%d\n", c);
    return 0;
}
```
