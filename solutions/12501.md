# Id 12501 — Flower Sticks - Bouquet

```c
#include <stdio.h>
#include <stdlib.h>

int cmpA(const void *a, const void *b) { return (*(int *)a - *(int *)b); }
int cmpD(const void *a, const void *b) { return (*(int *)b - *(int *)a); }

int main() {
    int N, K, i;
    int a[101];
    scanf("%d", &N);
    for (i = 0; i < N; i++) scanf("%d", &a[i]);
    if (scanf("%d", &K) != 1) K = N / 2;
    qsort(a, K, sizeof(int), cmpA);
    qsort(a + K, N - K, sizeof(int), cmpD);
    for (i = 0; i < N; i++) {
        if (i) printf(" ");
        printf("%d", a[i]);
    }
    printf("\n");
    return 0;
}
```
