#include <stdio.h>

int main() {
    int N;
    scanf("%d", &N);
    long long a[1000];
    int i;
    for (i = 0; i < N; i++) scanf("%lld", &a[i]);
    int k;
    for (k = 1; k <= N; k++) {
        long long sum = 0, mx;
        for (i = 0; i < k; i++) sum += a[i];
        mx = sum;
        for (i = k; i < N; i++) {
            sum += a[i] - a[i - k];
            if (sum > mx) mx = sum;
        }
        if (k > 1) printf(" ");
        printf("%lld", mx);
    }
    printf("\n");
    return 0;
}
