#include <stdio.h>
#include <string.h>
void reverseSecondHalf(char s[]) {
    int len = (int)strlen(s);
    for (int i = len / 2, j = len - 1; i < j; i++, j--) {
        char t = s[i]; s[i] = s[j]; s[j] = t;
    }
}
int main() {
    char s[101];
    scanf("%s", s);
    reverseSecondHalf(s);
    printf("%s\n", s);
    return 0;
}
