# Id 14515 — Even Frequency Sum

```c
#include <stdio.h>
#include <string.h>

int main() {
    char s[1001];
    scanf("%s", s);
    int L = strlen(s);
    long long sum = 0;
    int i = 0;
    while (i < L) {
        int j = i;
        while (j + 1 < L && s[j + 1] == s[i]) j++;
        int len = j - i + 1;
        if (len % 2 == 0) sum += len;
        i = j + 1;
    }
    printf("%lld\n", sum);
    return 0;
}
```
