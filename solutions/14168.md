#include <stdio.h>
#include <string.h>

int main() {
    char s[101];
    scanf("%s", s);
    int i, len = strlen(s);
    long long sum = 0;
    i = 0;
    while (i < len) {
        if (s[i] >= '0' && s[i] <= '9') {
            long long val = 0;
            while (i < len && s[i] >= '0' && s[i] <= '9') {
                val = val * 10 + (s[i] - '0');
                i++;
            }
            sum += val;
        } else {
            i++;
        }
    }
    printf("%lld\n", sum);
    return 0;
}
