# Id 12269 — Bunch of Room Keys

```c
#include <stdio.h>
#include <string.h>

int main() {
    char tok[10];
    char keys[100000][8];
    int total = 0, blanks = 0, uniq = 0;
    while (scanf("%s", tok) == 1) {
        if (tok[0] == 'q' || tok[0] == 'Q') break;
        total++;
        if (strcmp(tok, "-") == 0) {
            blanks++;
        } else {
            int dup = 0;
            for (int j = 0; j < uniq; j++) {
                if (strcmp(keys[j], tok) == 0) { dup = 1; break; }
            }
            if (!dup) strcpy(keys[uniq++], tok);
        }
    }
    printf("Blank Keys: %d\n", blanks);
    printf("Total Keys: %d\n", total);
    printf("Number of rooms: %d\n", uniq);
    return 0;
}
```
