```c
int getMinimumMoves(int N, int *arr) {
    long long sum = 0;
    for (int i = 0; i < N; i++) {
        sum += arr[i];
    }
    long long avg = sum / N;
    long long moves = 0;
    for (int i = 0; i < N; i++) {
        if (arr[i] > avg) {
            moves += arr[i] - avg;
        }
    }
    return (int)moves;
}
```
