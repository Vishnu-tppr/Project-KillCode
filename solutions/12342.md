# Id 12342 — Smallest Integer - Product of Digits

```c
#include <stdio.h>

int main() {
    long long n;
    scanf("%lld", &n);
    if (n == 1) {
        printf("1\n");
        return 0;
    }
    int digits[64], d = 0;
    int dd;
    for (dd = 9; dd >= 2; dd--) {
        while (n % dd == 0) {
            digits[d++] = dd;
            n /= dd;
        }
    }
    if (n != 1) {
        printf("-1\n");
        return 0;
    }
    int i, j;
    for (i = 0; i < d; i++)
        for (j = i + 1; j < d; j++)
            if (digits[j] < digits[i]) {
                int t = digits[i];
                digits[i] = digits[j];
                digits[j] = t;
            }
    for (i = 0; i < d; i++)
        printf("%d", digits[i]);
    printf("\n");
    return 0;
}
```
