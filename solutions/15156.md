# Id 15156 — N Queries - Find Occurrences of X

```c
#include <stdio.h>

int main() {
    int M, N, X;
    int a[10000], b[10000];
    int pos[10001];
    scanf("%d", &M);
    int i;
    for (i = 0; i < M; i++) scanf("%d", &a[i]);
    scanf("%d", &N);
    for (i = 0; i < N; i++) scanf("%d", &b[i]);
    scanf("%d", &X);
    int cnt = 0;
    for (i = 0; i < M; i++)
        if (a[i] == X) pos[cnt++] = i + 1;
    for (i = 0; i < N; i++) {
        if (i) printf(" ");
        if (b[i] >= 1 && b[i] <= cnt) printf("%d", pos[b[i] - 1]);
        else printf("-1");
    }
    printf("\n");
    return 0;
}
```
