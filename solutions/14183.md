# Id 14183 — Count 2s - Prime Factors

```c
#include <stdio.h>
#include <stdlib.h>

int main() {
    long long N;
    scanf("%lld", &N);
    long long t = N, rev = 0;
    while (t) { rev = rev * 10 + t % 10; t /= 10; }
    long long d = llabs(N - rev);
    int cnt = 0;
    while (d && d % 2 == 0) { cnt++; d /= 2; }
    printf("%d\n", cnt);
    return 0;
}
```
