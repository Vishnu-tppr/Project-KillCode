#include <stdio.h>
#include <string.h>

typedef struct {
    char name[25];
    double h;
} Stu;

int main() {
    int T;
    scanf("%d", &T);
    Stu s[35];
    int n = 0;
    char nm[25];
    double h;
    while (1) {
        if (scanf("%s", nm) != 1) break;
        if (nm[0] == 'q' || nm[0] == 'Q') break;
        scanf("%lf", &h);
        strcpy(s[n].name, nm);
        s[n].h = h;
        n++;
    }
    int idx[35];
    for (int i = 0; i < n; i++) idx[i] = i;
    for (int i = 0; i < n; i++) {
        for (int j = i + 1; j < n; j++) {
            if (s[idx[j]].h > s[idx[i]].h ||
                (s[idx[j]].h == s[idx[i]].h && idx[j] < idx[i])) {
                int t = idx[i]; idx[i] = idx[j]; idx[j] = t;
            }
        }
    }
    int sel[35], m = 0;
    for (int i = 0; i < T; i++) sel[m++] = idx[i];
    for (int i = 0; i < m; i++) {
        for (int j = i + 1; j < m; j++) {
            if (sel[j] < sel[i]) {
                int t = sel[i]; sel[i] = sel[j]; sel[j] = t;
            }
        }
    }
    for (int i = 0; i < m; i++) printf("%s\n", s[sel[i]].name);
    return 0;
}
