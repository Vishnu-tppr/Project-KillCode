# Id 2616 — Tower Line of Sight Issue

```c
#include <stdio.h>
int onSeg(long long ax, long long ay, long long bx, long long by, long long px, long long py) {
    long long cross = (bx-ax)*(py-ay) - (by-ay)*(px-ax);
    if (cross != 0) return 0;
    if (px < ax && px < bx) return 0;
    if (px > ax && px > bx) return 0;
    if (py < ay && py < by) return 0;
    if (py > ay && py > by) return 0;
    return 1;
}
int main() {
    long long ax, ay, bx, by, cx, cy, dx, dy;
    scanf("%lld%lld%lld%lld%lld%lld%lld%lld", &ax,&ay,&bx,&by,&cx,&cy,&dx,&dy);
    if (onSeg(ax,ay,cx,cy,bx,by) || onSeg(ax,ay,cx,cy,dx,dy) ||
        onSeg(bx,by,dx,dy,ax,ay) || onSeg(bx,by,dx,dy,cx,cy))
        printf("yes\n");
    else
        printf("no\n");
    return 0;
}
```
