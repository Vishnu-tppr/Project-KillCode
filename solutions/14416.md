#include <stdio.h>

int main() {
    int N, i;
    scanf("%d", &N);
    long long sum = 0, x;
    for (i = 0; i < N; i++) {
        scanf("%lld", &x);
        sum += x;
        if (i) printf(" ");
        printf("%lld", sum);
    }
    printf("\n");
    return 0;
}
