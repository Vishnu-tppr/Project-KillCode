# Id 12182 — Trains - Departure Time

```c
#include <stdio.h>

int main() {
    int N, i, h, m;
    int xh[100], xm[100];
    scanf("%d", &N);
    for (i = 0; i < N; i++) scanf("%d:%d", &xh[i], &xm[i]);
    scanf("%d:%d", &h, &m);
    for (i = 0; i < N; i++) {
        if (i) printf(" ");
        int cur = h * 60 + m;
        int y = cur + (xh[i] * 60 + xm[i]) - (xh[0] * 60 + xm[0]);
        printf("%02d:%02d", (y / 60) % 24, y % 60);
    }
    printf("\n");
    return 0;
}
```
