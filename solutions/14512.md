# Id 14512 — Count Even Sum Subarray

```c
#include <stdio.h>

int main() {
    int N, K;
    scanf("%d", &N);
    long long arr[1000];
    int i;
    for (i = 0; i < N; i++)
        scanf("%lld", &arr[i]);
    scanf("%d", &K);
    long long sum = 0;
    int count = 0;
    for (i = 0; i < K; i++)
        sum += arr[i];
    if (sum % 2 == 0)
        count++;
    for (i = K; i < N; i++) {
        sum = sum - arr[i - K] + arr[i];
        if (sum % 2 == 0)
            count++;
    }
    if (count > 0)
        printf("%d\n", count);
    else
        printf("-1\n");
    return 0;
}
```
