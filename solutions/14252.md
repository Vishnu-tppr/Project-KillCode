#include <math.h>

float maximumTravellingTime(int input1, int input2, int input3[][2])
{
    double maxTime = 0.0;
    int i;
    for (i = 1; i < input2; i++) {
        double dx = input3[i][0] - input3[i - 1][0];
        double dy = input3[i][1] - input3[i - 1][1];
        double dist = sqrt(dx * dx + dy * dy);
        double t = dist / input1;
        if (t > maxTime)
            maxTime = t;
    }
    return (float)maxTime;
}
