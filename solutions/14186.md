# Id 14186 — Steps & Jumps Game

```c
#include <stdio.h>

int main() {
    int N;
    scanf("%d", &N);
    int jumps = 0;
    while (N > 0) {
        if (N % 2 == 0) {
            N /= 2;
        } else {
            N -= 1;
        }
        jumps++;
    }
    printf("%d\n", jumps);
    return 0;
}
```
