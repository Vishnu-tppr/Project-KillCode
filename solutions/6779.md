int isSameReflection(char *str1, char *str2)
{
    int l1 = 0, l2 = 0;
    while (str1[l1]) l1++;
    while (str2[l2]) l2++;
    if (l1 != l2) return -1;
    for (int s = 0; s < l1; s++) {
        int ok = 1;
        for (int i = 0; i < l1; i++) {
            if (str1[(s + i) % l1] != str2[i]) { ok = 0; break; }
        }
        if (ok) return 1;
    }
    return -1;
}
