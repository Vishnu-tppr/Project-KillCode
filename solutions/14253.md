char *replaceAlternateWords(char *str) {
    static char res[2000];
    int i = 0, k = 0, word = 0;
    int first = 1;
    while (str[i]) {
        int start = i;
        while (str[i] && str[i] != '.') i++;
        if (!first) res[k++] = '.';
        first = 0;
        if (word % 2 == 0) {
            int j;
            for (j = start; j < i; j++) res[k++] = str[j];
        } else {
            res[k++] = 'x';
            res[k++] = 'y';
            res[k++] = 'z';
        }
        if (str[i] == '.') i++;
        word++;
    }
    res[k] = '\0';
    return res;
}
