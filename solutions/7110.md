# Id 7110 — Minimum Difference - N Integers

```c
#include <stdio.h>

int rev(int n) {
    int r = 0;
    while (n) { r = r * 10 + n % 10; n /= 10; }
    return r;
}

int main() {
    int n;
    scanf("%d", &n);
    int a[100];
    for (int i = 0; i < n; i++) { int x; scanf("%d", &x); a[i] = rev(x); }
    int min = 2000000000;
    for (int i = 0; i < n; i++)
        for (int j = i + 1; j < n; j++) {
            int d = a[i] - a[j];
            if (d < 0) d = -d;
            if (d < min) min = d;
        }
    printf("%d\n", min);
    return 0;
}

```
