# Id 12188 — Car Parking - Park & Search

```c
#include <stdio.h>
#include <string.h>

int main() {
    int N, M, Q, X, i;
    char reg[101][13];
    char R[13];
    scanf("%d", &N);
    for (i = 0; i < N; i++) scanf("%s", reg[i]);
    scanf("%d%d", &M, &Q);
    while (Q--) {
        scanf("%d%s", &X, R);
        if (X == 1) {
            if (N >= M) {
                printf("No space for parking\n");
            } else {
                strcpy(reg[N], R);
                N++;
                printf("Car parked at position: %d\n", N);
            }
        } else if (X == 2) {
            int found = 0;
            for (i = 0; i < N; i++) {
                if (strcmp(reg[i], R) == 0) {
                    printf("Car position: %d\n", i + 1);
                    found = 1;
                    break;
                }
            }
            if (!found) printf("Car does not exists\n");
        } else {
            printf("Invalid\n");
        }
    }
    return 0;
}
```
