# Id 14366 — Minimum MaxMin Difference

```c
#include <stdio.h>
#include <stdlib.h>

int cmp(const void *a, const void *b) {
    return *(const int *)a - *(const int *)b;
}

int main() {
    int N, M, i;
    scanf("%d %d", &N, &M);
    int a[100];
    for (i = 0; i < N; i++) scanf("%d", &a[i]);
    qsort(a, N, sizeof(int), cmp);
    int best = 1000000000;
    for (i = 0; i + M - 1 < N; i++) {
        int d = a[i + M - 1] - a[i];
        if (d < best) best = d;
    }
    printf("%d\n", best);
    return 0;
}
```
