# Id 14498 — All Combinations - Odd or Even

```c
#include <stdio.h>

int main() {
    int n, i, v;
    scanf("%d", &n);
    int odd = 0;
    for (i = 0; i < n; i++) {
        scanf("%d", &v);
        if (v % 2 != 0) odd++;
    }
    if (odd % 2 != 0)
        printf("ODD\n");
    else
        printf("EVEN\n");
    return 0;
}
```
