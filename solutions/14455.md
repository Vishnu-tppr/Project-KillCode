int isPerfectPalindrome(int N)
{
    if (N < 0) return 0;
    int r = 0, t = N;
    while (t) {
        r = r * 10 + t % 10;
        t /= 10;
    }
    if (r != N) return 0;
    long long s = 1;
    while (s * s < N) s++;
    return (s * s == N) ? 1 : 0;
}
