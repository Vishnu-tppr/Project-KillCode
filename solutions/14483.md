int customSpellCheck(char *str1, char *str2, int K) {
    int mismatch = 0;
    for (int i = 0; str1[i] != '\0' && str2[i] != '\0'; i++) {
        if (str1[i] != str2[i]) mismatch++;
    }
    if (mismatch == 0) return 1;
    if (mismatch < K) return 2;
    if (mismatch == K) return 3;
    return 4;
}
