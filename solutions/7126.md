# Id 7126 — Same Element - Two Arrays

```c
#include <stdio.h>

int main() {
    int n;
    scanf("%d", &n);
    int a[100], b[100];
    for (int i = 0; i < n; i++) scanf("%d", &a[i]);
    for (int i = 0; i < n; i++) scanf("%d", &b[i]);
    int f = 0;
    for (int i = 0; i < n; i++) {
        if (a[i] == b[n-1-i]) {
            if (f) printf(" ");
            printf("%d", a[i]);
            f = 1;
        }
    }
    if (!f) printf("-1");
    printf("\n");
    return 0;
}

```
