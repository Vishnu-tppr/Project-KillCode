# Id 6728 — Animal(s) or Bird(s) Sounds

```c
#include <stdio.h>
int main(){
    int n; char c; int i;
    scanf("%d", &n);
    for(i=0;i<n;i++){scanf(" %c", &c);
        if(c=='D')printf("Bark\n");
        else if(c=='C')printf("Meow\n");
        else if(c=='B')printf("Cuckoo\n");
    }
    return 0;
}
```
