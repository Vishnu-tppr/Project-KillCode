# Id 6381 — C - Function - Matrix Transpose

```c
#include <stdio.h>
void printTranspose(int N, int m[][100]) {
    for (int i = 0; i < N; i++) {
        for (int j = 0; j < N; j++) {
            if (j) printf(" ");
            printf("%d", m[j][i]);
        }
        printf("\n");
    }
}
int main() {
    int n, m[100][100];
    scanf("%d", &n);
    for (int i = 0; i < n; i++)
        for (int j = 0; j < n; j++) scanf("%d", &m[i][j]);
    printTranspose(n, m);
    return 0;
}
```
