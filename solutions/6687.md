int isLeapYear(int y)
{
    return (y % 400 == 0) || (y % 4 == 0 && y % 100 != 0);
}

long long daysFromEpoch(Date d)
{
    int monthDays[12] = {31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31};
    long long total = (d.year - 1) * 365LL + (d.year - 1) / 4 - (d.year - 1) / 100 + (d.year - 1) / 400;
    int m;
    for (m = 0; m < d.month - 1; m++)
        total += monthDays[m];
    if (d.month > 2 && isLeapYear(d.year))
        total += 1;
    total += d.date;
    return total;
}

int difference_in_dates(Date *date1, Date *date2)
{
    long long d1 = daysFromEpoch(*date1);
    long long d2 = daysFromEpoch(*date2);
    long long diff = d1 - d2;
    if (diff < 0)
        diff = -diff;
    return (int)diff;
}
