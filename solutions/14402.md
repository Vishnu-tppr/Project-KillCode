#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdlib.h>

int main() {
    int items[4] = {101, 102, 103, 104};
    double price[4] = {42, 50, 500, 40};
    int stock[4] = {10, 20, 15, 16};
    char a[32], b[32];
    while (scanf("%s", a) == 1) {
        int k;
        int isNum = 1;
        for (k = 0; a[k]; k++)
            if (!isdigit((unsigned char)a[k])) { isNum = 0; break; }
        if (!isNum) {
            printf("Invalid Input!\n");
            return 0;
        }
        int item = atoi(a);
        int idx = -1;
        for (k = 0; k < 4; k++)
            if (items[k] == item) { idx = k; break; }
        if (idx == -1) {
            printf("Invalid Input!\n");
            return 0;
        }
        if (scanf("%s", b) != 1) {
            printf("Invalid Input!\n");
            return 0;
        }
        isNum = 1;
        for (k = 0; b[k]; k++)
            if (!isdigit((unsigned char)b[k])) { isNum = 0; break; }
        if (!isNum) {
            printf("Invalid Input!\n");
            return 0;
        }
        int qty = atoi(b);
        if (qty < stock[idx]) {
            printf("%.1f INR\n", price[idx] * qty);
            stock[idx] -= qty;
            printf("%d LEFT\n", stock[idx]);
        } else {
            printf("NO STOCK\n%d LEFT\n", stock[idx]);
        }
    }
    return 0;
}
