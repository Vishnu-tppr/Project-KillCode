# Id 14418 — Cubic Bricks

```c
#include <stdio.h>

int main() {
    int N, v;
    scanf("%d", &N);
    int c = 0;
    for (int i = 0; i < N; i++) {
        scanf("%d", &v);
        for (long long r = 1; r * r * r <= v; r++) {
            if (r * r * r == v) { c++; break; }
        }
    }
    printf("%d\n", c);
    return 0;
}
```
