# Id 2611 — Odd Length String Diagonal Pattern [ZOHO]

```c
#include <stdio.h>
#include <string.h>
int main() {
    char s[52];
    int l, r, c;
    fgets(s, sizeof(s), stdin);
    s[strcspn(s, "\n")] = '\0';
    l = strlen(s);
    for (r = 0; r < l; r++) {
        for (c = 0; c < l; c++) {
            if (r == c || r + c == l - 1)
                printf("%c", s[c]);
            else
                printf(" ");
        }
        printf("\n");
    }
    return 0;
}
```
