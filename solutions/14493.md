# Id 14493 — Choose Blue and Black Pens

```c
#include <stdio.h>

long long C(int n, int r) {
    if (r > n - r) r = n - r;
    long long res = 1;
    int i;
    for (i = 1; i <= r; i++) res = res * (n - r + i) / i;
    return res;
}

int main() {
    int X, Y, M, N;
    scanf("%d %d %d %d", &X, &Y, &M, &N);
    printf("%lld\n", C(X, M) * C(Y, N));
    return 0;
}
```
