long long findNextTerm(int SIZE, int *arr)
{
    int i;

    if (SIZE == 1) return arr[0];

    int isAP = 1;
    long long d = (long long)arr[1] - arr[0];
    for (i = 1; i + 1 < SIZE; i++)
        if ((long long)arr[i + 1] - arr[i] != d) { isAP = 0; break; }
    if (isAP)
        return (long long)arr[SIZE - 1] + d;

    if (arr[0] != 0 && SIZE >= 2)
    {
        int isGP = 1;
        for (i = 0; i + 2 < SIZE; i++)

        if (isGP)
        {
            double r = (double)arr[1] / arr[0];
            return (long long)((double)arr[SIZE - 1] * r + 0.5);
        }
    }

    for (i = 2; i < SIZE; i++)
        if ((long long)arr[i] != (long long)arr[i - 1] + (long long)arr[i - 2]) { isFib = 0; break; }
    if (isFib)
        return (long long)arr[SIZE - 1] + (long long)arr[SIZE - 2];

    return -1;
}
