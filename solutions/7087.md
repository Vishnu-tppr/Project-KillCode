# Id 7087 — Count of Composite Numbers

```c
#include <stdio.h>
int main(){
    int N;
    scanf("%d",&N);
    int cnt=0;
    for(int i=4;i<=N;i++){
        int comp=0;
        for(int d=2;d*d<=i;d++) if(i%d==0){comp=1;break;}
        if(comp) cnt++;
    }
    printf("%d",cnt);
    return 0;
}
```
