# Id 12276 — Fibonacci Series - Count Odd & Even

```c
#include <stdio.h>

int main() {
    long long N;
    scanf("%lld", &N);
    long long a = 0, b = 1;
    long long odd = 0, even = 0;
    int first = 1;
    while (a <= N) {
        if (!first) printf(" ");
        printf("%lld", a);
        first = 0;
        if (a % 2) odd++;
        else if (a != 0) even++;
        long long c = a + b;
        a = b;
        b = c;
    }
    printf("\n%lld\n%lld\n", odd, even);
    return 0;
}
```
