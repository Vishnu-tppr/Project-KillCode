int findSubarrayUniqueCountSum(int N, int *arr, int K) {
    int total = 0;
    int s;
    for (s = 0; s + K <= N; s++) {
        int freq[1001] = {0};
        int i, uniq = 0;
        for (i = s; i < s + K; i++) freq[arr[i]]++;
        for (i = 0; i <= 1000; i++)
            if (freq[i] == 1) uniq++;
        total += uniq;
    }
    return total;
}
