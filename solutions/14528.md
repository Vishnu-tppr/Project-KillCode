#include <stdio.h>
#include <string.h>

int main() {
    char s[101];
    if (!fgets(s, sizeof s, stdin)) s[0] = 0;
    s[strcspn(s, "\n")] = 0;
    int L = strlen(s);
    while (L > 0 && s[L - 1] == ' ') s[--L] = 0;
    int valid = 1;
    if (L == 0) valid = 0;
    if (valid && !(s[0] >= 'A' && s[0] <= 'Z')) valid = 0;
    int dots = 0;
    for (int i = 0; i < L; i++)
        if (s[i] == '.') dots++;
    if (valid && s[L - 1] != '.') valid = 0;
    if (valid && dots != 1) valid = 0;
    int words = 0, in = 0;
    for (int i = 0; i < L; i++) {
        if (s[i] == ' ') in = 0;
        else if (!in) { in = 1; words++; }
    }
    if (valid && words < 2) valid = 0;
    if (valid) printf("True\n"); else printf("False\n");
    return 0;
}
