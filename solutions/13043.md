# Id 13043 — File - Characters at Odd Positions

```c
#include <stdio.h>
#include <string.h>

int main() {
    char f[100]; scanf("%s", f);
    FILE *fp = fopen(f, "r");
    char line[2000];
    while (fgets(line, sizeof(line), fp)) {
        int len = strlen(line);
        if (len && line[len - 1] == '\n') len--;
        for (int i = 0; i < len; i++)
            if ((i + 1) % 2 == 1) putchar(line[i]);
        putchar('\n');
    }
    fclose(fp);
    return 0;
}
```
